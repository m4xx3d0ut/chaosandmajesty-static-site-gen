---
- name: Deploy Chaos & Majesty site
  hosts: vps
  become: true
  gather_facts: true
  vars:
    image_repository: "{{ image_registry ~ (('/' ~ image_namespace) if (image_namespace | length) > 0 else '') }}"
    image_ref: "{{ image_repository }}/{{ image_name }}:{{ image_tag }}"
  pre_tasks:
    - name: Check Docker Engine availability
      command: docker --version
      changed_when: false

    - name: Check Docker Compose v2 availability
      command: docker compose version
      changed_when: false
  roles:
    - role: app_deploy
