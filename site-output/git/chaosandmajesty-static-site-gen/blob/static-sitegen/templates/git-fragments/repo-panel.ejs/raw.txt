<%
  const logContext = log || {};
  const logMode = logContext.mode || (logContext.isTruncated ? 'limited' : 'full');
  const fullFragment = logContext.fullFragmentPath || '';
  const limitedFragment = logContext.limitedFragmentPath || '';
  const totalCommits = Number.isFinite(logContext.totalCommits)
    ? logContext.totalCommits
    : (stats && Number.isFinite(stats.totalCommits) ? stats.totalCommits : 0);
  const displayedCommits = stats && Number.isFinite(stats.displayedCommits)
    ? stats.displayedCommits
    : (stats && Number.isFinite(stats.totalCommits) ? stats.totalCommits : 0);
  const initialLimit = Number.isFinite(logContext.initialLimit) ? logContext.initialLimit : displayedCommits;
  const showFullLink = !!(logContext && logContext.isTruncated && fullFragment);
  const showCollapseLink = !showFullLink && logMode === 'full' && limitedFragment && initialLimit > 0 && totalCommits > initialLimit;
%>
<article
  class="git-panel"
  data-git-panel="<%= repo.id %>"
  data-git-log-mode="<%= logMode %>"
  data-git-full-fragment="<%= fullFragment %>"
  data-git-limited-fragment="<%= limitedFragment %>"
  data-git-log-total="<%= totalCommits %>"
  data-git-log-initial-limit="<%= initialLimit %>"
>
  <header class="git-panel-header">
    <div class="git-panel-lead">
      <h4 class="git-panel-title"><%= repo.label %></h4>
      <p class="git-panel-meta">
        <% if (repo.owner) { %>
          <span class="git-panel-owner">@<%= repo.owner %></span>
        <% } %>
        <% if (repo.branch) { %>
          <span class="git-panel-branch">
            branch <strong><%= repo.branch %></strong>
          </span>
        <% } %>
      </p>
      <% if (repo.description) { %>
        <p class="git-panel-description"><%= repo.description %></p>
      <% } %>
    </div>
    <div class="git-panel-clone">
      <span class="git-clone-label">Clone repository (read-only)</span>
      <code class="git-clone-command" data-git-command="<%= clone.command %>"><%= clone.command %></code>
      <button type="button" class="git-clone-copy" data-copy-source="previous">Copy</button>
      <% if (showFullLink) { %>
        <a class="git-panel-log-link" data-git-log-action="expand" href="<%= fullFragment %>">Open full log</a>
      <% } else if (showCollapseLink) { %>
        <a class="git-panel-log-link" data-git-log-action="collapse" href="<%= limitedFragment %>">Show latest <%= initialLimit %></a>
      <% } %>
    </div>
  </header>

  <section class="git-activity">
    <header class="git-activity-header">
      <h5 class="git-activity-title">Recent commits</h5>
      <% if (stats && Number.isFinite(stats.displayedCommits)) { %>
        <span class="git-activity-count">
          <%= stats.displayedCommits %> shown
          <% if (logContext && logContext.isTruncated && totalCommits > stats.displayedCommits) { %>
            · of <%= totalCommits %> total
          <% } %>
        </span>
      <% } else if (stats && Number.isFinite(stats.totalCommits)) { %>
        <span class="git-activity-count"><%= stats.totalCommits %> shown</span>
      <% } %>
    </header>

    <% if (commits && commits.length > 0) { %>
      <ul class="git-commit-list">
        <% commits.forEach(function(commit) { %>
          <li class="git-commit-item">
            <div class="git-commit-summary">
              <% if (commit.commitUrl) { %>
                <a class="git-commit-title" href="<%= commit.commitUrl %>" target="_blank" rel="noopener"><%= commit.subject %></a>
              <% } else { %>
                <span class="git-commit-title"><%= commit.subject %></span>
              <% } %>
              <% if (commit.tags && commit.tags.length > 0) { %>
                <span class="git-commit-tags"><%= commit.tags.join(', ') %></span>
              <% } %>
            </div>
            <div class="git-commit-meta">
              <span class="git-commit-sha"><%= commit.shortSha %></span>
              <span class="git-commit-author"><%= commit.authorName %></span>
              <span class="git-commit-date" title="<%= commit.authoredIso %>">
                <%= commit.authoredDisplay %>
                <% if (commit.relativeTime) { %> · <%= commit.relativeTime %><% } %>
              </span>
            </div>
          </li>
        <% }); %>
      </ul>
    <% } else { %>
      <p class="git-commit-empty">No commits were detected for this branch.</p>
    <% } %>
  </section>
</article>
