       if (!value) {
         return depthPrefix || (hasAbsoluteBase ? (normalizedRootBase || '/') : './');
       }
       if (/^(\.\/|\.\.\/)/.test(value)) {
         return value;
       }
       const cleaned = value.replace(/^\/+/, '');
       if (hasAbsoluteBase) {
         return cleaned ? `${normalizedRootBase}/${cleaned}` : (normalizedRootBase || '/');
       }
       return depthPrefix + cleaned;
     }; %>
  <% const buildInternalHref = (href) => {
       if (!href) return '#';
       if (isExternal(href)) return href;
       if (href.startsWith('#')) {
         const target = href.slice(1);
         if (!target) {
           return resolvePath('');
         }
         return resolvePath(`sections/${target}.html`);
       }
       return resolvePath(href);
     }; %>
  <% const buildAssetHref = (asset) => {
       if (!asset) return '';
       if (isExternal(asset)) return asset;

  <script>
    document.addEventListener('DOMContentLoaded', function() {
      var toggleBtn = document.querySelector('.nav-toggle');
      var headerEl = document.querySelector('header');
      if (toggleBtn) {
        toggleBtn.addEventListener('click', function() {
          headerEl.classList.toggle('nav-open');
        });
      }
    });

    var filteredListItems = [];
    var currentSlug = null;

    function syncActiveLink(slug) {
      currentSlug = slug || null;
      document.querySelectorAll('.blog-post-link').forEach(function(link) {
        if (slug && link.dataset.postSlug === slug) {
          link.classList.add('is-active');
        } else {
          link.classList.remove('is-active');
        }
      });
      updateArticleNav();
    }

    function scrollArticleToTop() {
      var shell = document.querySelector('.blog-article-shell');
      if (shell && shell.scrollIntoView) {
        try {
          shell.scrollIntoView({ behavior: 'smooth', block: 'start' });
          return;
        } catch (error) {
          // continue to fallback if smooth scrolling is unsupported
        }
        shell.scrollIntoView(true);
        return;
      }
      if (typeof window !== 'undefined') {
        try {
          window.scrollTo({ top: 0, behavior: 'smooth' });
        } catch (error) {
          window.scrollTo(0, 0);
        }
      }
    }

    var sidebar = document.querySelector('.blog-sidebar');
    var filterButtons = Array.from(document.querySelectorAll('.author-filter-btn'));
    var searchInput = document.querySelector('.post-search-input');
    var postItems = Array.from(document.querySelectorAll('[data-post-item]'));
    var paginations = Array.from(document.querySelectorAll('.blog-sidebar-pagination'));
    var prevButtons = [];
    var nextButtons = [];
    paginations.forEach(function(pagination) {
      var prev = pagination.querySelector('[data-sidebar-prev]');
      var next = pagination.querySelector('[data-sidebar-next]');
      if (prev) {
        prevButtons.push(prev);
      }
      if (next) {
        nextButtons.push(next);
      }
    });
    var pageSize = sidebar ? parseInt(sidebar.dataset.pageSize || '6', 10) || 6 : 6;
    var currentAuthor = 'all';
    var currentSearch = '';
    var currentPage = 0;

    function normaliseText(value) {
      return (value || '').toString().toLowerCase();
    }

    function itemMatchesFilters(item) {
      var isHidden = item.dataset.hidden === 'true';
      var title = item.dataset.title || normaliseText(item.querySelector('h3') && item.querySelector('h3').textContent);
      var summary = item.dataset.summary || normaliseText(item.querySelector('p') && item.querySelector('p').textContent);
      var searchMatch = !currentSearch || title.indexOf(currentSearch) !== -1 || summary.indexOf(currentSearch) !== -1;

      if (isHidden) {
        return currentSearch && searchMatch;
      }

      if (!searchMatch) {
        return false;
      }

      var authorMatch = currentAuthor === 'all' || item.dataset.author === currentAuthor;
      return authorMatch;
    }

    function updatePaginationControls(totalPages, visibleCount) {
      if (!paginations.length) return;
      paginations.forEach(function(pagination) {
        if (visibleCount <= pageSize) {
          pagination.classList.add('is-hidden');
        } else {
          pagination.classList.remove('is-hidden');
        }
        var prevBtn = pagination.querySelector('[data-sidebar-prev]');
        var nextBtn = pagination.querySelector('[data-sidebar-next]');
        if (prevBtn) {
          prevBtn.disabled = currentPage <= 0;
        }
        if (nextBtn) {
          nextBtn.disabled = currentPage >= totalPages - 1;
        }
      });
    }

    function buildNavNeighborFromItem(item) {
      if (!item) return null;
      var link = item.querySelector('.blog-post-link');
      if (!link) return null;
      var titleNode = link.querySelector('h3');
      var titleText = '';
      if (titleNode && titleNode.textContent) {
        titleText = titleNode.textContent.trim();
      } else if (link.textContent) {
        titleText = link.textContent.trim();
      }
      return {
        slug: link.dataset.postSlug || '',
        title: titleText,
        href: link.getAttribute('href') || '#',
        fragmentHref: link.getAttribute('hx-get') || link.dataset.fragmentHref || ''
      };
    }

    function setNavLink(link, neighbor, direction) {
      if (!link) return;
      var fallbackTitle = direction === 'previous' ? 'No newer posts' : 'No older posts';
      var titleEl = link.querySelector('.blog-article-nav-title');

      if (!neighbor || !neighbor.slug) {
        link.classList.add('is-disabled');
        link.setAttribute('aria-disabled', 'true');
        link.setAttribute('tabindex', '-1');
        ['hx-get', 'hx-target', 'hx-swap', 'hx-push-url', 'hx-indicator'].forEach(function(attr) {
          link.removeAttribute(attr);
        });
        link.dataset.targetSlug = '';
        link.dataset.postSlug = '';
        link.dataset.fragmentHref = '';
        link.setAttribute('data-fragment-href', '');
        link.setAttribute('href', '#');
        if (titleEl) {
          titleEl.textContent = fallbackTitle;
        }
        return;
      }

      link.classList.remove('is-disabled');
      link.removeAttribute('aria-disabled');
      link.removeAttribute('tabindex');
      link.dataset.targetSlug = neighbor.slug;
      link.dataset.postSlug = neighbor.slug;
      link.dataset.fragmentHref = neighbor.fragmentHref || '';
      link.setAttribute('data-fragment-href', neighbor.fragmentHref || '');
      link.setAttribute('href', neighbor.href || '#');
      if (neighbor.fragmentHref) {
        link.setAttribute('hx-get', neighbor.fragmentHref);
      } else {
        link.removeAttribute('hx-get');
      }
      link.setAttribute('hx-target', '#blog-article');
      link.setAttribute('hx-swap', 'innerHTML');
      link.setAttribute('hx-push-url', '?post=' + neighbor.slug);
      link.setAttribute('hx-indicator', '#blog-loading');
      if (titleEl) {
        titleEl.textContent = neighbor.title || fallbackTitle;
      }
    }

    function updateArticleNav() {
      var articleContainer = document.getElementById('blog-article');
      if (!articleContainer) {
        return;
      }

      var articleNode = articleContainer.querySelector('[data-slug]');
      if (articleNode && !currentSlug) {
        currentSlug = articleNode.getAttribute('data-slug');
      }
      if (!articleNode) {
        currentSlug = null;
      }

      var prevLink = articleContainer.querySelector('[data-blog-nav="previous"]');
      var nextLink = articleContainer.querySelector('[data-blog-nav="next"]');
      if (!prevLink && !nextLink) {
        return;
      }

      var hasActiveFilter = currentAuthor !== 'all' || Boolean(currentSearch);
      var candidateItems = filteredListItems.length || hasActiveFilter ? filteredListItems : postItems;
      if (!currentSlug) {
        setNavLink(prevLink, null, 'previous');
        setNavLink(nextLink, null, 'next');
        return;
      }

      var currentIndex = -1;
      for (var i = 0; i < candidateItems.length; i += 1) {
        var item = candidateItems[i];
        if (!item) continue;
        var link = item.querySelector('.blog-post-link');
        if (link && link.dataset.postSlug === currentSlug) {
          currentIndex = i;
          break;
        }
      }

      if (currentIndex === -1) {
        setNavLink(prevLink, null, 'previous');
        setNavLink(nextLink, null, 'next');
        return;
      }

      var prevNeighbor = null;
      for (var j = currentIndex - 1; j >= 0; j -= 1) {
        prevNeighbor = buildNavNeighborFromItem(candidateItems[j]);
        if (prevNeighbor) break;
      }

      var nextNeighbor = null;
      for (var k = currentIndex + 1; k < candidateItems.length; k += 1) {
        nextNeighbor = buildNavNeighborFromItem(candidateItems[k]);
        if (nextNeighbor) break;
      }

      setNavLink(prevLink, prevNeighbor, 'previous');
      setNavLink(nextLink, nextNeighbor, 'next');
    }

    function applyFilters() {
      var filteredItems = [];
      postItems.forEach(function(item) {
        var matches = itemMatchesFilters(item);
        var isHidden = item.dataset.hidden === 'true';
        item.classList.toggle('is-hidden-filter', !matches);
        item.classList.toggle('is-search-visible', matches && isHidden);
        item.classList.remove('is-hidden-page');
        if (matches) {
          filteredItems.push(item);
        }
      });

      filteredListItems = filteredItems.slice();

      var totalPages = Math.max(1, Math.ceil(filteredItems.length / pageSize));
      if (currentPage >= totalPages) {
        currentPage = totalPages - 1;
      }
      var start = currentPage * pageSize;
      var end = start + pageSize;

      filteredItems.forEach(function(item, idx) {
        var inRange = idx >= start && idx < end;
        item.classList.toggle('is-hidden-page', !inRange);
      });

      updatePaginationControls(totalPages, filteredItems.length);
      updateArticleNav();
    }

    if (filterButtons.length) {
      filterButtons.forEach(function(btn) {
        btn.addEventListener('click', function() {
          var authorId = btn.dataset.authorFilter;
          currentAuthor = authorId || 'all';
          currentPage = 0;
          filterButtons.forEach(function(b) {
            b.classList.toggle('is-active', b === btn);
          });
          applyFilters();
        });
      });
    }

    if (searchInput) {
      searchInput.addEventListener('input', function(event) {
        currentSearch = normaliseText(event.target.value).trim();
        currentPage = 0;
        applyFilters();
      });
    }

    if (prevButtons.length) {
      prevButtons.forEach(function(btn) {
        btn.addEventListener('click', function() {
          if (currentPage > 0) {
            currentPage -= 1;
            applyFilters();
          }
        });
      });
    }

    if (nextButtons.length) {
      nextButtons.forEach(function(btn) {
        btn.addEventListener('click', function() {
          currentPage += 1;
          applyFilters();
        });
      });
    }

    applyFilters();

    var params = new URLSearchParams(window.location.search || '');
    var initialSlug = params.get('post');
    var articleEl = document.getElementById('blog-article');

    if (initialSlug && window.htmx && articleEl) {
      var initialLink = document.querySelector('[data-post-slug="' + initialSlug + '"]');
      if (initialLink && initialLink.getAttribute('hx-get')) {
        syncActiveLink(initialSlug);
        htmx.ajax('GET', initialLink.getAttribute('hx-get'), {
          target: '#blog-article',
          swap: 'innerHTML',
          history: false
        });
      }
    } else {
      var initialArticle = articleEl && articleEl.querySelector('[data-slug]');
      if (initialArticle) {
        syncActiveLink(initialArticle.getAttribute('data-slug'));
      }
    }

    document.addEventListener('htmx:beforeRequest', function(event) {
      var trigger = event.detail && event.detail.elt;
      if (!trigger || !trigger.matches || (!trigger.matches('.blog-post-link') && !trigger.matches('.blog-article-nav-link'))) {
        return;
      }
      syncActiveLink(trigger.dataset.postSlug);
      var article = document.getElementById('blog-article');
      if (article) {
        article.setAttribute('aria-busy', 'true');
      }
    });

    document.addEventListener('htmx:afterSwap', function(event) {
      if (event.target && event.target.id === 'blog-article') {
        var articleNode = event.target.querySelector('[data-slug]');
        if (articleNode) {
          syncActiveLink(articleNode.getAttribute('data-slug'));
        }
        event.target.setAttribute('aria-busy', 'false');
        scrollArticleToTop();
      }
    });

    document.addEventListener('htmx:historyRestore', function() {
      var articleNode = document.querySelector('#blog-article [data-slug]');
      if (articleNode) {
        syncActiveLink(articleNode.getAttribute('data-slug'));
      } else {
        syncActiveLink(null);
      }
      var article = document.getElementById('blog-article');
      if (article) {
        article.setAttribute('aria-busy', 'false');
      }
    });
  </script>
