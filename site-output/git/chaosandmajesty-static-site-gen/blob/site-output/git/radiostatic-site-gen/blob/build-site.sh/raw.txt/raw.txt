#!/usr/bin/env bash

set -euo pipefail

CONFIG_DEFAULT="${RADIOSTATIC_CONFIG:-${SITEGEN_CONFIG:-smoke-test.yaml}}"
OUTPUT_DEFAULT="${RADIOSTATIC_OUTPUT:-${SITEGEN_OUTPUT:-site-output}}"
VERBOSE_DEFAULT="${RADIOSTATIC_VERBOSE:-${SITEGEN_VERBOSE:-1}}"

CONFIG_PATH="${1:-$CONFIG_DEFAULT}"
OUTPUT_DIR="${2:-$OUTPUT_DEFAULT}"
VERBOSE_FLAG="$VERBOSE_DEFAULT"

if [ ! -f "$CONFIG_PATH" ]; then
    echo "Error: configuration file '$CONFIG_PATH' not found" >&2
    exit 1
fi

echo "Building site with RadioStatic..."

if [[ "${ENABLE_DOOM_BUILD:-0}" != "0" ]]; then
    if [[ -x "scripts/build-doom.sh" ]]; then
        echo "Preparing DOOM wasm assets (ENABLE_DOOM_BUILD enabled)..."
        if ! ./scripts/build-doom.sh; then
            echo "Warning: DOOM assets could not be prepared. Continuing build without them." >&2
        fi
    else
        echo "Warning: scripts/build-doom.sh missing; skipping DOOM asset build." >&2
    fi
fi

RADIOSTATIC_ARGS=("-c" "$CONFIG_PATH" "-o" "$OUTPUT_DIR")
if [[ "$VERBOSE_FLAG" == "1" || "$VERBOSE_FLAG" == "true" ]]; then
    RADIOSTATIC_ARGS+=("-v")
fi

./static-sitegen/bin/radiostatic.js "${RADIOSTATIC_ARGS[@]}"

if [[ ! -f "$OUTPUT_DIR/index.html" ]]; then
    echo "Error: index.html not generated in $OUTPUT_DIR" >&2
    exit 1
fi

if [[ ! -d "$OUTPUT_DIR/assets" ]]; then
    echo "Error: assets directory not copied to $OUTPUT_DIR" >&2
    exit 1
fi

echo "RadioStatic build completed successfully!"
echo "Config: $CONFIG_PATH"
echo "Output: $OUTPUT_DIR"
find "$OUTPUT_DIR" -type f | head -10
