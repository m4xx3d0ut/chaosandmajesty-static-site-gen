#!/usr/bin/env python3
"""Annotate existing Markdown code fences with language hints.

The script splits inline ```prompt fences onto their own lines and
heuristically upgrades unfixed fences that contain console output.
"""
from __future__ import annotations

import argparse
from pathlib import Path

PROMPT_PREFIXES = (
    "┌", "└", "kali@", "operator@", "www-data@", "root@",
    "PS ", "C:", "mimikatz #",
)
HTTP_PREFIXES = ("GET ", "POST ", "HEAD ", "PUT ", "DELETE ", "OPTIONS ", "HTTP/")
POWERSHELL_HINTS = (
    "PS ", "Directory:", "Mode ", "Volume ", "Windows IP Configuration",
    "Ethernet adapter", "IPv4 Address", "Subnet Mask", "Default Gateway", "C:",
)


def normalize_inline_fences(lines: list[str]) -> tuple[list[str], bool]:
    normalized: list[str] = []
    changed = False
    for line in lines:
        stripped = line.lstrip()
        if stripped.startswith("```"):
            rest = stripped[3:]
            if rest and not rest[0].isalpha():
                candidate = rest.lstrip()
                if candidate:
                    normalized.append("```")
                    normalized.append(candidate)
                    changed = True
                    continue
        normalized.append(line)
    return normalized, changed


def guess_language(content: str) -> str | None:
    stripped = content.lstrip()
    for prefix in HTTP_PREFIXES:
        if stripped.startswith(prefix):
            return "http"
    for hint in POWERSHELL_HINTS:
        if stripped.startswith(hint):
            return "powershell"
    for prefix in PROMPT_PREFIXES:
        if stripped.startswith(prefix):
            return "bash"
    if "@" in stripped and "$" in stripped:
        return "bash"
    if stripped.startswith(("sudo ", "curl ", "python", "ffuf ", "nc ", "hydra ", "hashcat ", "nmap ", "xfreerdp", "sqlmap", "ssh ")):
        return "bash"
    return None


def annotate_fences(path: Path) -> bool:
    lines = path.read_text(encoding="utf-8").splitlines()
    result, normalized = normalize_inline_fences(lines)
    changed = normalized
    lines = result

    i = 0
    while i < len(lines):
        line = lines[i]
        stripped = line.strip()
        if stripped.startswith("```"):
            lang = stripped[3:].strip()
            if not lang:
                j = i + 1
                while j < len(lines) and not lines[j].strip():
                    j += 1
                if j < len(lines):
                    language = guess_language(lines[j])
                    if language:
                        lines[i] = f"```{language}"
                        changed = True
            i += 1
            continue
        i += 1

    if changed:
        path.write_text("\n".join(lines) + "\n", encoding="utf-8")
    return changed


def main(files: list[str]) -> None:
    changed_any = False
    for name in files:
        path = Path(name)
        if not path.is_file():
            print(f"Skipping missing file: {path}")
            continue
        if annotate_fences(path):
            print(f"Updated {path}")
            changed_any = True
    if not changed_any:
        print("No fences needed updates.")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Annotate existing Markdown code fences with language hints.")
    parser.add_argument("files", nargs="+", help="Markdown files to process")
    args = parser.parse_args()
    main(args.files)
