#!/usr/bin/env python3
"""Promote blog markdown bodies into their companion content files.

Copies the body (below the front matter) from each blog/<slug>.md into the
matching file under content/**/<slug>.md while preserving the content file's
existing front matter.
"""
from __future__ import annotations

import argparse
from pathlib import Path
from typing import Iterable

REPO_ROOT = Path(__file__).resolve().parent.parent
BLOG_DIR = REPO_ROOT / "blog"
CONTENT_DIR = REPO_ROOT / "content"


def split_front_matter(text: str) -> tuple[str, str]:
    lines = text.splitlines()
    if not lines or lines[0].strip() != "---":
        return "", text

    end = None
    for idx in range(1, len(lines)):
        if lines[idx].strip() == "---":
            end = idx
            break
    if end is None:
        return "", text

    fm = "\n".join(lines[: end + 1]) + "\n\n"
    body = "\n".join(lines[end + 1 :]).lstrip("\n")
    if body and not body.endswith("\n"):
        body += "\n"
    return fm, body


def strip_labs_sections(body: str) -> str:
    lines = body.splitlines()
    result = []
    skip = False
    for line in lines:
        stripped = line.lstrip()
        if not skip and stripped.startswith('###### Labs'):
            skip = True
            continue
        if skip and stripped.startswith('#'):
            skip = False
        if not skip:
            result.append(line)
    cleaned = '\n'.join(result).rstrip('\n')
    if cleaned:
        cleaned += '\n'
    return cleaned


def find_content_file(blog_file: Path) -> Path | None:
    matches = list(CONTENT_DIR.rglob(blog_file.name))
    if not matches:
        return None
    if len(matches) > 1:
        raise RuntimeError(
            f"Multiple content candidates for {blog_file.relative_to(REPO_ROOT)}: "
            + ", ".join(str(p.relative_to(REPO_ROOT)) for p in matches)
        )
    return matches[0]


def promote(files: Iterable[Path], overwrite: bool) -> None:
    for blog_path in files:
        content_path = find_content_file(blog_path)
        if content_path:
            if not overwrite:
                print(f"Skipping {blog_path.relative_to(REPO_ROOT)} (content exists; use --overwrite to replace)")
                continue
        else:
            print(f"Skipping {blog_path.relative_to(REPO_ROOT)} (no content match)")
            continue

        blog_body = split_front_matter(blog_path.read_text(encoding='utf-8'))[1]
        if 'content/offsec' in str(content_path.relative_to(REPO_ROOT).parent):
            blog_body = strip_labs_sections(blog_body)

        content_text = content_path.read_text(encoding="utf-8")
        content_fm, _ = split_front_matter(content_text)
        if not content_fm:
            # keep original file untouched if it lacks expected front matter
            print(
                f"Skipping {content_path.relative_to(REPO_ROOT)} (no front matter)")
            continue

        updated = content_fm + blog_body
        if not updated.endswith("\n"):
            updated += "\n"
        content_path.write_text(updated, encoding="utf-8")
        print(
            "Promoted",
            f"{blog_path.relative_to(REPO_ROOT)} -> {content_path.relative_to(REPO_ROOT)}",
        )


def main() -> None:
    parser = argparse.ArgumentParser(description="Promote blog markdown to content")
    parser.add_argument("paths", nargs="*", help="Specific blog markdown files to promote")
    parser.add_argument("--overwrite", action="store_true", help="Replace existing content files")
    args = parser.parse_args()

    if args.paths:
        blog_files = [BLOG_DIR / Path(p).name for p in args.paths]
    else:
        blog_files = sorted(BLOG_DIR.glob("*.md"))

    blog_files = [p for p in blog_files if p.is_file()]
    promote(blog_files, overwrite=args.overwrite)


if __name__ == "__main__":
    main()
