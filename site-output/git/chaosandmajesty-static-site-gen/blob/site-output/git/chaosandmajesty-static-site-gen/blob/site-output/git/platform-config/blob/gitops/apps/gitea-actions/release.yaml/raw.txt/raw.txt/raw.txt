apiVersion: helm.toolkit.fluxcd.io/v2
kind: HelmRelease
metadata:
  name: gitea-actions
  namespace: git-local
spec:
  releaseName: gitea-actions
  interval: 5m
  timeout: 15m
  install:
    remediation:
      retries: 3
  upgrade:
    remediation:
      retries: 3
  chart:
    spec:
      chart: .
      # version: "0.0.0"
      sourceRef:
        kind: GitRepository
        name: helm-actions
        namespace: flux-system
  values:
    enabled: true
    existingSecret: "gitea-runner-token"
    existingSecretKey: "GITEA_RUNNER_REGISTRATION_TOKEN"
    giteaRootURL: "https://gitea.core.home.arpa"

    statefulset:
      actRunner:
        env:
          - name: DOCKER_HOST
            value: "tcp://127.0.0.1:2376"
          - name: DOCKER_TLS_VERIFY
            value: "1"
          - name: DOCKER_CERT_PATH
            value: "/certs/client"
        extraVolumeMounts:
          - name: gitea-cert
            mountPath: /usr/local/share/ca-certificates/gitea-root.crt
            subPath: gitea-root.crt
            readOnly: true
        command:
          - sh
          - -c
          - |
            update-ca-certificates
            echo "Waiting for Docker daemon with TLS on localhost..."
            until timeout 10 docker info > /dev/null 2>&1; do
              echo "Docker not ready, retrying..."
              sleep 2
            done
            echo "Docker is ready, starting act-runner..."
            exec run.sh
        config: |
          log:
            level: debug
          cache:
            enabled: false
          runner:
            insecure: false
            docker_host: tcp://127.0.0.1:2376
          container:
            options: >
              -e DOCKER_HOST=tcp://docker:2376
              -e DOCKER_TLS_VERIFY=1
              -e DOCKER_CERT_PATH=/certs/client
              -e SSL_CERT_FILE=/usr/local/share/ca-certificates/gitea-root.crt
              -e NODE_EXTRA_CA_CERTS=/usr/local/share/ca-certificates/gitea-root.crt
              -e CURL_CA_BUNDLE=/usr/local/share/ca-certificates/gitea-root.crt
              -v /certs/client:/certs/client:ro
              -v /usr/local/share/ca-certificates/gitea-root.crt:/usr/local/share/ca-certificates/gitea-root.crt:ro
              --add-host=docker:host-gateway
            valid_volumes:
              - /certs/client
              - /usr/local/share/ca-certificates/gitea-root.crt
            docker_host: tcp://127.0.0.1:2376

      dind:
        tag: 27-dind
        env:
          - name: DOCKER_TLS_CERTDIR
            value: "/certs"
        command:
          - sh
          - -c
          - |
            update-ca-certificates
            exec dockerd-entrypoint.sh "$@"
          - --
        extraVolumeMounts:
          - name: gitea-cert
            mountPath: /usr/local/share/ca-certificates/gitea-root.crt
            subPath: gitea-root.crt
            readOnly: true
          - name: docker-daemon-config
            mountPath: /etc/docker/daemon.json
            subPath: daemon.json
            readOnly: true
        extraEnvs:
          - name: DOCKER_IPTABLES_LEGACY
            value: "1"
        args:
          - --host=tcp://0.0.0.0:2376
          - --host=unix:///var/run/docker.sock
          - --tlsverify
          - --tlscacert=/certs/server/ca.pem
          - --tlscert=/certs/server/cert.pem
          - --tlskey=/certs/server/key.pem
          - --insecure-registry=reg.microk8s.core.home.arpa:32000

      extraVolumes:
        - name: gitea-cert
          secret:
            secretName: gitea-tls
            items:
              - key: ca.crt
                path: gitea-root.crt
        - name: docker-daemon-config
          configMap:
            name: docker-daemon-config
