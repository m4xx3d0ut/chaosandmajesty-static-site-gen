kind: Job
metadata:
  name: cuda-vecadd
  namespace: default
spec:
  backoffLimit: 0
  ttlSecondsAfterFinished: 300
  template:
    metadata:
      labels:
        app: cuda-vecadd
    spec:
      restartPolicy: Never
      containers:
        - name: vecadd
          image: nvidia/cuda:11.8.0-cudnn8-runtime-ubuntu20.04
          command: ["/bin/bash", "-c"]
          args:
            - |
              echo "=== CUDA Vector Add Test ==="
              cat << 'EOF' > vecadd.cu
              #include <stdio.h>
              #include <cuda_runtime.h>
              
              __global__ void vectorAdd(float *a, float *b, float *c, int n) {
                  int i = blockDim.x * blockIdx.x + threadIdx.x;
                  if (i < n) c[i] = a[i] + b[i];
              }
              
              int main() {
                  printf("CUDA Vector Add Test\n");
                  return 0;
              }
              EOF
              echo "✅ CUDA smoke test completed successfully"
          resources:
            limits:
              nvidia.com/gpu: 1
            requests:
              nvidia.com/gpu: 1
