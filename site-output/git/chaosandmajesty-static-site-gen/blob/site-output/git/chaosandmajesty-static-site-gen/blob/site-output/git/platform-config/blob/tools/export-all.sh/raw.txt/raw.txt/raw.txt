#!/usr/bin/env bash
# Export live cluster state → files, optional Helm & CRDs, optional Gitea app.ini,
# move + SOPS-encrypt Secrets, print a summary, and clean .tmp on exit.
#
# Usage:
#   tools/export-all.sh [--helm] [--include-crds] [--gitea-app-ini -n <ns> --selector <k=v>] [--encrypt-secrets] [--keep-tmp]
#
# Examples:
#   tools/export-all.sh --encrypt-secrets
#   tools/export-all.sh --helm --include-crds --encrypt-secrets
#   tools/export-all.sh --gitea-app-ini -n gitea --selector app=gitea --encrypt-secrets
set -euo pipefail

# -------------------- config --------------------
KUBECTL="microk8s kubectl"
HELM="microk8s helm3"

OUT_ROOT="live-export"
CLUSTER_DIR="$OUT_ROOT/kubernetes/cluster-scoped"
NS_DIR="$OUT_ROOT/kubernetes/namespaces"
SECRETS_DIR="$OUT_ROOT/secrets"
HELM_DIR="$OUT_ROOT/helm"
APPS_DIR="$OUT_ROOT/apps"
TMP_DIR=".tmp"

DO_HELM=false
DO_CRDS=false
DO_APPINI=false
DO_ENCRYPT=false
KEEP_TMP=false
APP_NS=""
APP_SELECTOR=""

# Skip lists (reduce noise)
SKIP_CLUSTER_KINDS=( "events.events.k8s.io" "events" "componentstatuses" )
SKIP_NS_KINDS=( "events.events.k8s.io" "events" "endpoints" "endpointslices.discovery.k8s.io" "leases.coordination.k8s.io" "controllerrevisions.apps" )

# -------------------- args --------------------
while [[ $# -gt 0 ]]; do
  case "$1" in
    --helm) DO_HELM=true; shift;;
    --include-crds) DO_CRDS=true; shift;;
    --gitea-app-ini) DO_APPINI=true; shift;;
    -n|--namespace) APP_NS="${2:-}"; shift 2;;
    --selector) APP_SELECTOR="${2:-}"; shift 2;;
    --encrypt-secrets) DO_ENCRYPT=true; shift;;
    --keep-tmp) KEEP_TMP=true; shift;;
    -h|--help) sed -n '1,80p' "$0"; exit 0;;
    *) echo "Unknown flag: $1"; exit 1;;
  esac
done

# -------------------- deps --------------------
need() { command -v "$1" >/dev/null 2>&1 || { echo "Missing dep: $1"; exit 1; }; }
need jq
need yq
$DO_HELM && need microk8s >/dev/null 2>&1 || true
if $DO_ENCRYPT; then need sops; fi

# -------------------- setup / cleanup --------------------
mkdir -p "$TMP_DIR" "$CLUSTER_DIR" "$NS_DIR" "$SECRETS_DIR" "$HELM_DIR" "$APPS_DIR"
cleanup() { $KEEP_TMP || rm -rf "$TMP_DIR"; }
trap cleanup EXIT

# yq filter for list output
strip_list_noise() {
  yq eval 'del(.items[].metadata.managedFields)
           | del(.items[].status)
           | del(.items[].metadata.annotations."kubectl.kubernetes.io/last-applied-configuration")' -
}
# jq filter for single objects
strip_obj_noise_jq='del(
  .metadata.managedFields,
  .status,
  .metadata.resourceVersion,
  .metadata.uid,
  .metadata.generation,
  .metadata.creationTimestamp,
  .metadata.annotations."kubectl.kubernetes.io/last-applied-configuration"
)'

is_skipped() { local k="$1"; shift; local arr=("$@"); for s in "${arr[@]}"; do [[ "$k" == "$s" ]] && return 0; done; return 1; }

# -------------------- export: cluster-scoped --------------------
echo "=== Cluster-scoped resources ==="
for r in $($KUBECTL api-resources --verbs=list --namespaced=false -o name); do
  if is_skipped "$r" "${SKIP_CLUSTER_KINDS[@]}"; then echo "Skip: $r"; continue; fi
  echo "Kind: $r"
  f="$TMP_DIR/cluster-${r//\//-}.yaml"
  if $KUBECTL get "$r" -o yaml > "$f" 2>/dev/null; then
    <"$f" strip_list_noise > "${f}.clean" && mv "${f}.clean" "$f"
    count=$(yq eval '.items | length // 0' "$f")
    if [[ "$count" -gt 0 ]]; then
      yq -o=json '.items[]' "$f" | jq -c '.' | while IFS= read -r obj; do
        [[ -z "${obj:-}" ]] && continue
        kind=$(jq -r '.kind // "unknown"' <<<"$obj")
        name=$(jq -r '.metadata.name // "unnamed"' <<<"$obj")
        out="$CLUSTER_DIR/${kind,,}/${name}.yaml"
        mkdir -p "$(dirname "$out")"
        jq "$strip_obj_noise_jq" <<<"$obj" | yq -P > "$out"
        echo "  -> $out"
      done
    fi
  fi
done

# -------------------- export: namespaced --------------------
echo "=== Namespaced resources ==="
namespaces=$($KUBECTL get ns -o jsonpath='{.items[*].metadata.name}')
for r in $($KUBECTL api-resources --verbs=list --namespaced -o name); do
  if is_skipped "$r" "${SKIP_NS_KINDS[@]}"; then echo "Skip: $r"; continue; fi
  echo "Kind: $r"
  for ns in $namespaces; do
    f="$TMP_DIR/ns-${ns}-${r//\//-}.yaml"
    if $KUBECTL -n "$ns" get "$r" -o yaml > "$f" 2>/dev/null; then
      <"$f" strip_list_noise > "${f}.clean" && mv "${f}.clean" "$f"
      count=$(yq eval '.items | length // 0' "$f")
      if [[ "$count" -gt 0 ]]; then
        yq -o=json '.items[]' "$f" | jq -c '.' | while IFS= read -r obj; do
          [[ -z "${obj:-}" ]] && continue
          kind=$(jq -r '.kind // "unknown"' <<<"$obj")
          name=$(jq -r '.metadata.name // "unnamed"' <<<"$obj")
          # Skip autogenerated SA token secrets
          if [[ "$kind" == "Secret" ]]; then
            stype=$(jq -r '.type // ""' <<<"$obj")
            [[ "$stype" == "kubernetes.io/service-account-token" ]] && continue
          fi
          out="$NS_DIR/$ns/${kind,,}/${name}.yaml"
          mkdir -p "$(dirname "$out")"
          jq "$strip_obj_noise_jq" <<<"$obj" | yq -P > "$out"
          echo "  -> $out"
        done
      fi
    fi
  done
done

# -------------------- export: CRDs (optional) --------------------
if $DO_CRDS; then
  echo "=== CRDs ==="
  mkdir -p "$CLUSTER_DIR/crds"
  $KUBECTL get crds -o yaml \
    | yq eval 'del(.items[].metadata.managedFields) | del(.items[].status)' - \
    > "$CLUSTER_DIR/crds/all-crds.yaml"
  echo "  -> $CLUSTER_DIR/crds/all-crds.yaml"
fi

# -------------------- export: Helm (optional) --------------------
if $DO_HELM; then
  echo "=== Helm releases ==="
  if ! command -v microk8s >/dev/null 2>&1; then echo "WARN: microk8s not found; skipping Helm."; else
    mapfile -t helm_namespaces < <($HELM list -A | awk 'NR>1{print $2}' | sort -u)
    for ns in "${helm_namespaces[@]}"; do
      mapfile -t releases < <($HELM -n "$ns" list | awk 'NR>1{print $1}')
      for rel in "${releases[@]}"; do
        base="$HELM_DIR/$ns/$rel"; mkdir -p "$base"
        $HELM -n "$ns" get values "$rel" > "$base/values.yaml" || true
        $HELM -n "$ns" get values "$rel" --all > "$base/values-all.yaml" || true
        $HELM -n "$ns" get manifest "$rel" > "$base/manifest.yaml" || true
        echo "  -> $base/"
      done
    done
  fi
fi

# -------------------- export: Gitea app.ini (optional) --------------------
if $DO_APPINI; then
  echo "=== Gitea app.ini ==="
  if [[ -z "$APP_NS" || -z "$APP_SELECTOR" ]]; then
    echo "ERR: --gitea-app-ini requires -n <ns> and --selector <key=val>"; exit 1
  fi
  POD=$($KUBECTL -n "$APP_NS" get pod -l "$APP_SELECTOR" -o jsonpath='{.items[0].metadata.name}' 2>/dev/null || true)
  if [[ -n "${POD:-}" ]]; then
    mkdir -p "$APPS_DIR/gitea"
    $KUBECTL -n "$APP_NS" exec "$POD" -- cat /data/gitea/conf/app.ini > "$APPS_DIR/gitea/app.ini" || true
    echo "  -> $APPS_DIR/gitea/app.ini"
  else
    echo "  -> No pod matched selector '$APP_SELECTOR' in ns '$APP_NS'"
  fi
fi

# -------------------- move + SOPS-encrypt Secrets (optional) --------------------
if $DO_ENCRYPT; then
  echo "=== Move + SOPS encrypt Secrets ==="
  moved=0
  while IFS= read -r -d '' f; do
    ns=$(echo "$f" | awk -F'/namespaces/' '{print $2}' | awk -F'/' '{print $1}')
    name=$(basename "$f" .yaml)
    out="$SECRETS_DIR/$ns/$name.yaml"
    mkdir -p "$(dirname "$out")"
    mv "$f" "$out"
    ((moved++)) || true
  done < <(find "$NS_DIR" -type f -path '*/secret/*' -name '*.yaml' -print0)

  if [[ "$moved" -gt 0 ]]; then
    echo "Moved $moved Secret files into $SECRETS_DIR"
    find "$SECRETS_DIR" -type f -name '*.yaml' -print0 \
      | xargs -0 -I{} sops --encrypt --in-place {}
    echo "Encrypted Secrets with sops."
  else
    echo "No Secrets to move/encrypt."
  fi
fi

# -------------------- summary --------------------
echo "=== Summary ==="
total_files=$(find "$OUT_ROOT" -type f -name '*.yaml' | wc -l | tr -d ' ')
echo "YAML files exported: $total_files"

echo "Top kinds (cluster-scoped):"
find "$CLUSTER_DIR" -type f -name '*.yaml' | awk -F'/' '{print $(NF-1)}' \
  | sort | uniq -c | sort -nr | head -10 || true

echo "Top kinds (namespaced):"
find "$NS_DIR" -type f -name '*.yaml' | awk -F'/' '{print $(NF-2)}' \
  | sort | uniq -c | sort -nr | head -10 || true

echo "Per-namespace counts:"
find "$NS_DIR" -type f -name '*.yaml' | awk -F'/' '{print $(NF-3)}' \
  | sort | uniq -c | sort -nr || true

echo "Done."
