# k8s/base/vllm/starcoder2-15b.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: vllm-starcoder2-15b
  namespace: ai
  labels: { app.kubernetes.io/name: vllm-starcoder2-15b, app.kubernetes.io/part-of: ai-system, ai.zlabs.dev/component: vllm }
  annotations:
    kustomize.toolkit.fluxcd.io/ignore: "true"
    ai.zlabs.dev/maxReplicas: "1"
spec:
  replicas: 0
  selector:
    matchLabels: { app.kubernetes.io/name: vllm-starcoder2-15b, app.kubernetes.io/part-of: ai-system }
  template:
    metadata:
      labels: { app.kubernetes.io/name: vllm-starcoder2-15b, app.kubernetes.io/part-of: ai-system, ai.zlabs.dev/component: vllm }
    spec:
      enableServiceLinks: false
      runtimeClassName: nvidia
      nodeSelector: { accelerator: nvidia }
      terminationGracePeriodSeconds: 1200
      containers:
        - name: vllm
          image: vllm/vllm-openai:latest-x86_64
          env:
            - name: VLLM_USE_V1
              value: "0"
            - name: VLLM_ATTENTION_BACKEND
              value: XFORMERS            # <-- UPPERCASE
            - name: VLLM_USE_FLASHINFER
              value: "0"
            - name: VLLM_USE_FLASHINFER_SAMPLER
              value: "0"
            - name: HUGGING_FACE_HUB_TOKEN
              valueFrom:
                secretKeyRef:
                  name: hf-token
                  key: HUGGING_FACE_HUB_TOKEN
                  optional: true
            - name: HF_HUB_ENABLE_HF_TRANSFER
              value: "1"
          args:
            - "--model=bigcode/starcoder2-15b-instruct-v0.1"
            - "--port=8000"
            - "--trust-remote-code"
            - "--max-model-len=8192"
            - "--gpu-memory-utilization=0.85"
            - "--swap-space=4"
            - "--download-dir=/root/.cache/huggingface"
          ports:
            - { name: http, containerPort: 8000 }

          # Allow long cold-start without restarts
          startupProbe:
            httpGet: { path: /health, port: http }
            periodSeconds: 5
            timeoutSeconds: 2
            failureThreshold: 600

          # Ready only when API lists models
          readinessProbe:
            httpGet: { path: /v1/models, port: http }
            initialDelaySeconds: 10
            periodSeconds: 5
            timeoutSeconds: 2
            failureThreshold: 120

          # Very forgiving liveness during/after load
          livenessProbe:
            httpGet: { path: /health, port: http }
            initialDelaySeconds: 1200
            periodSeconds: 15
            timeoutSeconds: 3
            failureThreshold: 3

          resources:
            requests: { cpu: "8", memory: "32Gi", nvidia.com/gpu: "1" }
            limits:   { cpu: "16", memory: "96Gi", nvidia.com/gpu: "1" }

          volumeMounts:
            - { name: hf-cache, mountPath: /root/.cache/huggingface }
      volumes: [ { name: hf-cache, persistentVolumeClaim: { claimName: hf-cache } } ]
---
apiVersion: v1
kind: Service
metadata:
  name: vllm-starcoder2-15b
  namespace: ai
  labels: { app.kubernetes.io/name: vllm-starcoder2-15b }
spec:
  selector: { app.kubernetes.io/name: vllm-starcoder2-15b }
  ports:
    - { name: openai, port: 8000, targetPort: 8000 }
    - { name: cachectl, port: 8081, targetPort: 8081 }
