(function() {
  const HISTORY_KEY = 'radiostatic-console-history';
  const LEGACY_HISTORY_KEYS = ['sitegen-demo-history'];
  const THEME_KEY = 'radiostatic-theme-mode';
  const LEGACY_THEME_KEYS = ['sitegen-theme-mode'];
  const PROMPT = 'rs>';
  const READY_MESSAGE = 'signal ready';

  function readAndAdopt(key, legacyKeys) {
    try {
      const existing = localStorage.getItem(key);
      if (existing) {
        return existing;
      }
      for (const legacy of legacyKeys) {
        const legacyValue = localStorage.getItem(legacy);
        if (legacyValue) {
          localStorage.setItem(key, legacyValue);
          return legacyValue;
        }
      }
    } catch (error) {
      console.warn('RadioStatic storage unavailable:', error);
    }
    return null;
  }

  const history = JSON.parse(readAndAdopt(HISTORY_KEY, LEGACY_HISTORY_KEYS) || '[]');
  let historyIndex = history.length;

  function saveHistory(entry) {
    history.push(entry);
    if (history.length > 50) {
      history.shift();
    }
    historyIndex = history.length;
    try {
      localStorage.setItem(HISTORY_KEY, JSON.stringify(history));
    } catch (error) {
      console.warn('RadioStatic could not persist console history:', error);
    }
  }

  function qs(selector) {
    return document.querySelector(selector);
  }

  function qsa(selector) {
    return Array.from(document.querySelectorAll(selector));
  }

  function initNavToggle() {
    const toggle = qs('.nav-toggle');
    const links = qs('.nav-links');
    if (!toggle || !links) return;

    const mq = window.matchMedia ? window.matchMedia('(max-width: 960px)') : null;

    function syncNav() {
      if (mq && mq.matches) {
        links.classList.remove('is-open');
        toggle.setAttribute('aria-expanded', 'false');
      } else {
        links.classList.add('is-open');
        toggle.setAttribute('aria-expanded', 'false');
      }
    }

    toggle.addEventListener('click', function() {
      const isOpen = links.classList.toggle('is-open');
      toggle.setAttribute('aria-expanded', isOpen ? 'true' : 'false');
    });

    if (mq) {
      if (typeof mq.addEventListener === 'function') {
        mq.addEventListener('change', syncNav);
      } else if (typeof mq.addListener === 'function') {
        mq.addListener(syncNav);
      }
      syncNav();
    } else {
      links.classList.add('is-open');
      toggle.setAttribute('aria-expanded', 'false');
    }
  }

  function resolveInitialMode() {
    const stored = readAndAdopt(THEME_KEY, LEGACY_THEME_KEYS);
    if (stored === 'dark' || stored === 'light') {
      return stored;
    }
    const body = document.body;
    if (body) {
      if (body.classList.contains('light-mode')) {
        return 'light';
      }
      if (body.classList.contains('dark-mode')) {
        return 'dark';
      }
      if (body.dataset && body.dataset.theme === 'light') {
        return 'light';
      }
    }
    return 'dark';
  }

  function applyTheme(mode) {
    const body = document.body;
    const toggle = qs('.theme-toggle');
    if (!body) return;

    const normalized = mode === 'light' ? 'light' : 'dark';
    body.classList.toggle('dark-mode', normalized === 'dark');
    body.classList.toggle('light-mode', normalized === 'light');
    body.classList.toggle('theme-dark', normalized === 'dark');
    body.classList.toggle('theme-light', normalized === 'light');

    if (toggle) {
      const nextLabel = normalized === 'dark' ? 'LIGHT' : 'DARK';
      toggle.textContent = nextLabel;
      toggle.setAttribute('aria-pressed', normalized === 'dark' ? 'true' : 'false');
      toggle.title = normalized === 'dark' ? 'Switch to light mode' : 'Switch to dark mode';
    }
    try {
      localStorage.setItem(THEME_KEY, normalized);
      LEGACY_THEME_KEYS.forEach(function(legacyKey) {
        localStorage.removeItem(legacyKey);
      });
    } catch (error) {
      console.warn('RadioStatic could not persist theme preference:', error);
    }
  }

  function initThemeToggle() {
    applyTheme(resolveInitialMode());

    const toggle = qs('.theme-toggle');
    if (!toggle) return;

    toggle.addEventListener('click', function() {
      const current = readAndAdopt(THEME_KEY, LEGACY_THEME_KEYS) || resolveInitialMode();
      applyTheme(current === 'dark' ? 'light' : 'dark');
    });
  }

  function initTerminal() {
    const output = qs('.console-frame pre');
    const input = qs('.console-input');
    const reset = qs('.console-reset');
    if (!output || !input) return;

    const commands = {
      help() {
        return [
          'Available commands:',
          '  help   - show this message',
          '  docs   - open the RadioStatic documentation index',
          '  build  - explain how to trigger a build',
          '  clear  - reset the console view'
        ].join('\n');
      },
      docs() {
        return 'Open README.md or run `npm run docs` (coming soon) to view the RadioStatic guides.';
      },
      build() {
        return 'Run `./build-site.sh smoke-test.yaml` to create the RadioStatic demo or pass your own YAML file.';
      },
      clear() {
        output.textContent = PROMPT + ' ' + READY_MESSAGE + '\\n';
        return null;
      }
    };

    function appendLine(text) {
      output.textContent += text + '\n';
      output.scrollTop = output.scrollHeight;
    }

    // seed the console
    if (!output.textContent.trim()) {
      output.textContent = PROMPT + ' ' + READY_MESSAGE + '\n';
    }

    input.addEventListener('keydown', function(event) {
      if (event.key === 'ArrowUp') {
        if (historyIndex > 0) {
          historyIndex -= 1;
          input.value = history[historyIndex] || '';
        }
        event.preventDefault();
        return;
      }
      if (event.key === 'ArrowDown') {
        if (historyIndex < history.length - 1) {
          historyIndex += 1;
          input.value = history[historyIndex] || '';
        } else {
          historyIndex = history.length;
          input.value = '';
        }
        event.preventDefault();
        return;
      }
      if (event.key !== 'Enter') {
        return;
      }

      const raw = input.value.trim();
      if (!raw) {
        appendLine(PROMPT);
        return;
      }

      appendLine(PROMPT + ' ' + raw);
      saveHistory(raw);
      input.value = '';

      const command = raw.toLowerCase();
      const handler = commands[command];
      if (handler) {
        const response = handler();
        if (typeof response === 'string') {
          appendLine(response);
        }
      } else {
        appendLine(`Unknown command: ${command}`);
        appendLine('Type `help` to see the built-ins.');
      }
    });

    if (reset) {
      reset.addEventListener('click', function() {
        input.value = '';
        output.textContent = PROMPT + ' ' + READY_MESSAGE + '\n';
        history.length = 0;
        historyIndex = 0;
        try {
          localStorage.removeItem(HISTORY_KEY);
          LEGACY_HISTORY_KEYS.forEach(function(legacyKey) {
            localStorage.removeItem(legacyKey);
          });
        } catch (error) {
          console.warn('RadioStatic could not clear console history:', error);
        }
      });
    }
  }

  function hydrateLatestArticles() {
    const map = window.radiostaticLatestArticles || window.sitegenLatestArticles || {};
    qsa('[data-author-latest]').forEach(function(node) {
      const authorId = node.getAttribute('data-author-latest');
      const info = map[authorId];
      if (!info) return;

      node.innerHTML = '';
      const link = document.createElement('a');
      link.href = info.href;
      link.textContent = info.title;
      node.appendChild(link);
    });
  }

  window.addEventListener('DOMContentLoaded', function() {
    initNavToggle();
    initThemeToggle();
    initTerminal();
    hydrateLatestArticles();
  });
})();
