### CA (Option 0)

Generate pem, key, crt.
```
openssl req -x509 -nodes -new -sha256 -days 1024 -newkey rsa:2048 -keyout localCA.key -out localCA.pem -subj "/C=US/CN=Local-Root-CA"
openssl x509 -outform pem -in localCA.pem -out localCA.crt
```

Create ext.
```
authorityKeyIdentifier=keyid,issuer
basicConstraints=CA:FALSE
keyUsage = digitalSignature, nonRepudiation, keyEncipherment, dataEncipherment
subjectAltName = @alt_names
[alt_names]
DNS.1 = localhost
DNS.2 = hyp3rion.local
DNS.3 = www.hyp3rion.local
```

Generate key, csr, and crt.
```
openssl req -new -nodes -newkey rsa:2048 -keyout local.key -out local.csr -subj "/C=US/ST=CA/L=HB/O=Local-Certificates/CN=hyp3rion.local"

openssl x509 -req -sha256 -days 1024 -in local.csr -CA $HOME/local_ssl/ca/root/localCA.pem -CAkey $HOME/local_ssl/ca/root/localCA.key -CAcreateserial -extfile hyp3rion.ext -out local.crt
```

**Import pem to Browser**
1. Chrome settings (Settings > Privacy and Security > Security > Manage certificates > Authorities > Import). Check Trust this certificate for identifying websites
2. Firefox: Preferences > Privacy and Security > Certificates > View Certificates > Authorities > import. Check Trust this CA to identify websites

### CA (Option 1, WiP so probably won't work without editing)

```bash
if [ "$EUID" -ne 0 ]
  then echo "Please run as root"
  exit
fi

export CA=hyp3rion
export NAME="hyp3rion.local"
export DAYS=3650

echo "----------------------------------------------------"
echo "[!] Once prompted, set FQDN to the value to: $NAME -"
echo "----------------------------------------------------"

# you probably want to have this in its own directory
mkdir /etc/ssl/$CA && cd /etc/ssl/$CA

# generate private key
openssl genrsa -des3 -out $CA.key 2048

# generate root certificate
openssl req -x509 -new -nodes -key $CA.key -sha256 -days 825 -out $CA.pem

# if CA files were in a separate directory
cd .. && mkdir /etc/ssl/$NAME && cd /etc/ssl/$NAME

# generate private key
openssl genrsa -out $NAME.key 2048

# Create a certificate-signing request
# Once prompted, set FQDN to the value of $NAME
openssl req -new -key $NAME.key -out $NAME.csr

# Create a config file for the extensions
>$NAME.ext cat <<-EOF
authorityKeyIdentifier=keyid,issuer
basicConstraints=CA:FALSE
keyUsage = digitalSignature, nonRepudiation, keyEncipherment, dataEncipherment
subjectAltName = @alt_names
[alt_names]
DNS.1 = www.$NAME
DNS.2 = $NAME
EOF

# Create the signed certificate
#openssl x509 -req -in $NAME.csr -CA $CA.pem -CAkey $CA.key -CAcreateserial -out $NAME.crt -days $DAYS -sha256 -extfile $NAME.ext

sudo openssl x509 -req -in $NAME.csr -CA /etc/ssl/$CA/$CA.pem -CAkey /etc/ssl/$CA/$CA.key -CAcreateserial -out $NAME.crt -days $DAYS -sha256 -extfile $NAME.ext
```

**Import pem to Browser**
1. Chrome settings (Settings > Privacy and Security > Security > Manage certificates > Authorities > Import). Check Trust this certificate for identifying websites

### Cert & Key (Option 2, self-signed but without CA Root Cert)

Create conf.
```
cd
mkdir local_ssl
cd local_ssl
touch ssl.conf
```

Edit `ssl.conf`.
```
[req]
distinguished_name = req_distinguished_name
x509_extensions = v3_req
prompt = no
[req_distinguished_name]
C = <--COUNTRY CODE - 2 Characters-->
ST = <--PROVINCE/ STATE-->
L = <--CITY-->
O = <--ORGANISATION-->
OU = <--DEPARTMENT-->
CN = <--CERTIFICATE ISSUER NAME - Can be anything-->
[v3_req]
keyUsage = nonRepudiation, digitalSignature, keyEncipherment
extendedKeyUsage = serverAuth
subjectAltName = @alt_names
[alt_names]
DNS.1 = <--DOMAIN NAME-->
DNS.2 = <--DOMAIN NAME 2-->
```

Create the cert.
```
openssl req -x509 -nodes -days 3650 -newkey rsa:2048 -keyout localhost.key -out localhost.crt -config ssl.conf -extensions 'v3_req'

openssl x509 -req -in hyp3rion.local.csr -CA .pem -CAkey $CA.key -CAcreateserial -out $NAME.crt -days 825 -sha256 -extfile $NAME.ext

openssl req -x509 -nodes -days 3650 -newkey rsa:2048 -keyout cert.key -out cert.crt -config req.conf -extensions v3_req
```

Copy the cert and key to the SSL directory.
```bash
sudo cp localhost.crt /etc/ssl/certs/localhost.crt
sudo cp localhost.key /etc/ssl/private/localhost.key
```


## Configure Nginx for SSL

Edit the Nginx conf to include.
```
server {
      listen 443 ssl;
      listen [::]:443 ssl;
      ssl_certificate /etc/ssl/certs/localhost.crt;
      ssl_certificate_key /etc/ssl/private/localhost.key;
      ssl_protocols TLSv1.2 TLSv1.1 TLSv1;
```
(or)
```
server {
      listen 443 ssl;
      listen [::]:443 ssl;
      ssl_certificate /etc/ssl/hyp3rion/hyp3rion.pem;
      ssl_certificate_key /etc/ssl/hyp3rion.key;
      ssl_protocols TLSv1.2 TLSv1.1 TLSv1;
```

(If desired) Redirect all HTTP requests to HTTPS.
```
server {
    listen 80;
    listen [::]:80;
    server_name <YOUR-SERVER-NAME>;
    return 301 https://$server_name$request_uri;
}
```

Test the Nginx conf and reload.
```bash
sudo nginx -T
sudo systemctl restart nginx
```


## Add Cert to Local DB

Install `libnss3-tools`.
```bash
sudo apt-get update
sudo apt-get install libnss3-tools
```

Add cert to the DB.
```bash
sudo cp localhost.crt /usr/share/ca-certificates/
certutil -d sql:$HOME/.pki/nssdb -A -t "CT,c,c" -n "localhost" -i /usr/share/ca-certificates/localhost.crt
```
