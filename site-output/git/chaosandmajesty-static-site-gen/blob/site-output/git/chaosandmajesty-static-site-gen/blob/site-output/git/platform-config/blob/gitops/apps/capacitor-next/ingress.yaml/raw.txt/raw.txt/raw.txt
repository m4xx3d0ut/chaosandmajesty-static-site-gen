# gitops/apps/capacitor-next/ingress.yaml
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: capacitor-next
  namespace: flux-system
  annotations:
    nginx.ingress.kubernetes.io/proxy-connect-timeout: "10"
    nginx.ingress.kubernetes.io/proxy-read-timeout: "60"
    nginx.ingress.kubernetes.io/auth-type: basic
    nginx.ingress.kubernetes.io/auth-secret: capacitor-next-auth
    nginx.ingress.kubernetes.io/auth-realm: 'Authentication Required'
spec:
  ingressClassName: nginx
  tls:
    - hosts: [capacitor-next.core.home.arpa]
      secretName: capacitor-next-tls
  rules:
    - host: capacitor-next.core.home.arpa
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: capacitor-next
                port:
                  number: 3333
