+static void DoomMusic_EnsureBuffer(size_t frames)
+{
+    size_t samples = frames * 2;
+    if (g_music_state.mix_buffer_capacity >= samples)
+    {
+        return;
+    }
+    size_t new_capacity = frames * 2;
+    int16_t *new_buffer = (int16_t *)realloc(g_music_state.mix_buffer, new_capacity * sizeof(int16_t));
+    if (!new_buffer)
+    {
+        I_Error("Failed to allocate DOOM music mix buffer");
+    }
+    g_music_state.mix_buffer = new_buffer;
+    g_music_state.mix_buffer_capacity = new_capacity;

+static uint8_t DoomMusic_ReadByte(const unsigned char **cursor, const unsigned char *end)
+{
+    if (*cursor >= end)
+    {
+        return 0;
+    }
+    uint8_t value = **cursor;
+    (*cursor)++;
+    return value;
+}
+
+static uint16_t DoomMusic_ReadLE16(const unsigned char **cursor, const unsigned char *end)
+{
+    uint16_t lo = DoomMusic_ReadByte(cursor, end);
+    uint16_t hi = DoomMusic_ReadByte(cursor, end);
+    return (hi << 8) | lo;
+}
+
+static uint32_t DoomMusic_ReadVarLen(const unsigned char **cursor, const unsigned char *end)
+{
+    uint32_t value = 0;
+    int shift = 0;
+    while (*cursor < end)
+    {
+        uint8_t byte = DoomMusic_ReadByte(cursor, end);
+        value |= (uint32_t)(byte & 0x7F) << shift;
+        if ((byte & 0x80) == 0)
+        {
+            break;
+        }
+        shift += 7;
+    }
+    return value;
+}

+static int DoomMusic_AddEvent(doom_music_event_t **events,
+                              size_t *count,
+                              size_t *capacity,
+                              doom_music_event_t event)
+{
+    if (*count >= *capacity)
+    {
+        size_t new_capacity = (*capacity == 0) ? 128 : (*capacity * 2);
+        doom_music_event_t *new_block = (doom_music_event_t *)realloc(*events, new_capacity * sizeof(doom_music_event_t));
+        if (!new_block)
+        {
+            return 0;
+        }
+        *events = new_block;
+        *capacity = new_capacity;
+    }
+    (*events)[*count] = event;
+    (*count)++;
+    return 1;
+}

+static int DoomMusic_Parse(const unsigned char *data, size_t length,
+                           doom_music_event_t **out_events,
+                           size_t *out_count,
+                           uint32_t *out_last_tick)
+{
+    const unsigned char *cursor = data;
+    const unsigned char *end = data + length;
+
+    if (length < 16)
+    {
+        return 0;
+    }
+
+    // MUS header fields
+    cursor += 4; // signature "MUS\x1A"
+    uint16_t score_len = DoomMusic_ReadLE16(&cursor, end);
+    uint16_t score_start = DoomMusic_ReadLE16(&cursor, end);
+    uint16_t channel_count = DoomMusic_ReadLE16(&cursor, end);
+    (void)channel_count;
+    cursor += 4; // skip secondary channels and instrument count
+
+    if (score_start >= length)
+    {
+        return 0;
+    }
+
+    const unsigned char *score = data + score_start;
+    const unsigned char *score_end = score + score_len;
+    if (score_end > end)
+    {
+        score_end = end;
+    }
+
+    doom_music_event_t *events = NULL;
+    size_t event_count = 0;
+    size_t event_capacity = 0;
+    uint32_t tick = 0;
+    uint8_t channel_map[16];
+    uint8_t channel_volume[16];
+    size_t i;
+
+    for (i = 0; i < 16; ++i)
+    {
+        channel_map[i] = i;
+        if (i == 15)
+        {
+            channel_map[i] = 9;
+        }
+        else if (i >= 9)
+        {
+            channel_map[i] = i + 1;
+        }
+        channel_volume[i] = 100;
+    }
+
+    cursor = score;
+    while (cursor < score_end)
+    {
+        uint8_t event_byte = DoomMusic_ReadByte(&cursor, score_end);
+        int last_event = (event_byte & 0x80) != 0;
+        uint8_t event_type = (event_byte >> 4) & 0x07;
+        uint8_t mus_channel = event_byte & 0x0F;
+        uint8_t midi_channel = channel_map[mus_channel & 0x0F];
+
+        switch (event_type)
+        {
+            case 0: // release note
+            {
+                uint8_t note = DoomMusic_ReadByte(&cursor, score_end) & 0x7F;
+                doom_music_event_t ev = { tick, 0, DM_EVENT_NOTE_OFF, midi_channel, note, 0 };
+                if (!DoomMusic_AddEvent(&events, &event_count, &event_capacity, ev))
+                {
+                    free(events);
+                    return 0;
+                }
+                break;
+            }
+            case 1: // play note
+            {
+                uint8_t note_data = DoomMusic_ReadByte(&cursor, score_end);
+                uint8_t note = note_data & 0x7F;
+                uint8_t volume = channel_volume[mus_channel];
+                if (note_data & 0x80)
+                {
+                    volume = DoomMusic_ReadByte(&cursor, score_end) & 0x7F;
+                    channel_volume[mus_channel] = volume;
+                }
+                if (volume == 0)
+                {
+                    volume = 1;
+                }
+                doom_music_event_t ev = { tick, 0, DM_EVENT_NOTE_ON, midi_channel, note, volume };
+                if (!DoomMusic_AddEvent(&events, &event_count, &event_capacity, ev))
+                {
+                    free(events);
+                    return 0;
+                }
+                break;
+            }
+            case 2: // pitch wheel
+            {
+                // ignore for now
+                DoomMusic_ReadByte(&cursor, score_end);
+                break;
+            }
+            case 3: // system event (ignore)
+            {
+                DoomMusic_ReadByte(&cursor, score_end);
+                break;
+            }
+            case 4: // controller change
+            {
+                uint8_t controller = DoomMusic_ReadByte(&cursor, score_end) & 0x7F;
+                uint8_t value = DoomMusic_ReadByte(&cursor, score_end) & 0x7F;
+                doom_music_event_t ev;
+                switch (controller)
+                {
+                    case 0: // instrument change
+                        ev.tick = tick;
+                        ev.sample = 0;
+                        ev.type = DM_EVENT_PROGRAM;
+                        ev.channel = midi_channel;
+                        ev.data1 = value;
+                        ev.data2 = 0;
+                        if (!DoomMusic_AddEvent(&events, &event_count, &event_capacity, ev))
+                        {
+                            free(events);
+                            return 0;
+                        }
+                        break;
+                    case 3: // volume
+                        channel_volume[mus_channel] = value;
+                        ev.tick = tick;
+                        ev.sample = 0;
+                        ev.type = DM_EVENT_VOLUME;
+                        ev.channel = midi_channel;
+                        ev.data1 = value;
+                        ev.data2 = 0;
+                        if (!DoomMusic_AddEvent(&events, &event_count, &event_capacity, ev))
+                        {
+                            free(events);
+                            return 0;
+                        }
+                        break;
+                    case 4: // pan
+                        ev.tick = tick;
+                        ev.sample = 0;
+                        ev.type = DM_EVENT_PAN;
+                        ev.channel = midi_channel;
+                        ev.data1 = value;
+                        ev.data2 = 0;
+                        if (!DoomMusic_AddEvent(&events, &event_count, &event_capacity, ev))
+                        {
+                            free(events);
+                            return 0;
+                        }
+                        break;
+                    default:
+                        break;
+                }
+                break;
+            }
+            case 5: // score end
+            {
+                doom_music_event_t ev = { tick, 0, DM_EVENT_END, midi_channel, 0, 0 };
+                if (!DoomMusic_AddEvent(&events, &event_count, &event_capacity, ev))
+                {
+                    free(events);
+                    return 0;
+                }
+                cursor = score_end;
+                break;
+            }
+            default:
+                break;
+        }
+
+        if (last_event)
+        {
+            uint32_t delta = DoomMusic_ReadVarLen(&cursor, score_end);
+            tick += delta;
+        }
+    }
+
+    *out_events = events;
+    *out_count = event_count;
+    *out_last_tick = tick;
+    return 1;
+}

+static doom_music_voice_t *DoomMusic_AllocVoice(void)
+{
+    size_t i;
+    doom_music_voice_t *voice = NULL;
+    float lowest_env = 1e9f;
+    for (i = 0; i < DOOM_MUSIC_MAX_VOICES; ++i)
+    {
+        if (!g_music_state.voices[i].active)
+        {
+            voice = &g_music_state.voices[i];
+            break;
+        }
+        if (g_music_state.voices[i].env < lowest_env)
+        {
+            lowest_env = g_music_state.voices[i].env;
+            voice = &g_music_state.voices[i];
+        }
+    }
+    if (voice)
+    {
+        voice->active = 1;
+        voice->releasing = 0;
+    }
+    return voice;
+}

+static float DoomMusic_NoteFrequency(int note)
+{
+    return 440.0f * (float)pow(2.0, ((double)note - 69.0) / 12.0);
+}
+
+static void DoomMusic_ProcessEvent(const doom_music_event_t *event)
+{
+    switch (event->type)
+    {
+        case DM_EVENT_NOTE_ON:
+        {
+            doom_music_voice_t *voice = DoomMusic_AllocVoice();
+            if (!voice)
+            {
+                return;
+            }
+            float freq = DoomMusic_NoteFrequency(event->data1);
+            voice->channel = event->channel;
+            voice->note = event->data1;
+            voice->velocity = (float)event->data2 / 127.0f;
+            if (voice->velocity <= 0.0f)
+            {
+                voice->velocity = 0.01f;
+            }
+            voice->phase = 0.0f;
+            voice->phase_step = (float)((2.0 * M_PI * freq) / (double)g_music_state.sample_rate);
+            voice->env = voice->velocity;
+            voice->release_factor = (float)pow(0.0001, 1.0 / (DOOM_MUSIC_RELEASE_SECONDS * g_music_state.sample_rate));
+            break;
+        }
+        case DM_EVENT_NOTE_OFF:
+        {
+            size_t i;
+            for (i = 0; i < DOOM_MUSIC_MAX_VOICES; ++i)
+            {
+                doom_music_voice_t *voice = &g_music_state.voices[i];
+                if (!voice->active)
+                {
+                    continue;
+                }
+                if (voice->channel == event->channel && voice->note == event->data1)
+                {
+                    voice->releasing = 1;
+                }
+            }
+            break;
+        }
+        case DM_EVENT_VOLUME:
+        {
+            g_music_state.channel_volume[event->channel & 0x0F] = (float)event->data1 / 127.0f;
+            break;
+        }
+        case DM_EVENT_PAN:
+        {
+            g_music_state.channel_pan[event->channel & 0x0F] = (float)event->data1 / 127.0f;
+            break;
+        }
+        case DM_EVENT_PROGRAM:
+        default:
+            break;
+    }
+}
+
+static void DoomMusic_ResetVoices(void)
+{
+    size_t i;
+    for (i = 0; i < DOOM_MUSIC_MAX_VOICES; ++i)
+    {
+        g_music_state.voices[i].active = 0;
+        g_music_state.voices[i].releasing = 0;
+        g_music_state.voices[i].env = 0.0f;
+    }
+}

+static int DoomMusic_Render(short *output, size_t frames)
+{
+    size_t frame_index = 0;
+    size_t out_index = 0;
+
+    while (frame_index < frames)
+    {
+        // process pending events whose sample time has arrived
+        while (g_music_state.next_event < g_music_state.event_count)
+        {
+            doom_music_event_t *event = &g_music_state.events[g_music_state.next_event];
+            if (event->sample > g_music_state.current_sample)
+            {
+                break;
+            }
+            if (event->type == DM_EVENT_END)
+            {
+                if (g_music_state.looping)
+                {
+                    g_music_state.current_sample = 0;
+                    g_music_state.next_event = 0;
+                    DoomMusic_ResetVoices();
+                    continue;
+                }
+                else
+                {
+                    g_music_state.playing = 0;
+                    return (int)frame_index;
+                }
+            }
+            DoomMusic_ProcessEvent(event);
+            g_music_state.next_event++;
+        }
+
+        size_t frames_to_render = frames - frame_index;
+        if (g_music_state.next_event < g_music_state.event_count)
+        {
+            doom_music_event_t *next_event = &g_music_state.events[g_music_state.next_event];
+            if (next_event->sample > g_music_state.current_sample)
+            {
+                uint64_t delta = next_event->sample - g_music_state.current_sample;
+                if (delta < frames_to_render)
+                {
+                    frames_to_render = (size_t)delta;
+                }
+            }
+            else
+            {
+                frames_to_render = 0;
+            }
+        }
+
+        if (frames_to_render == 0)
+        {
+            // no audio to render before next event, advance a minimal step
+            frames_to_render = 1;
+        }
+
+        size_t i;
+        for (i = 0; i < frames_to_render; ++i)
+        {
+            float left = 0.0f;
+            float right = 0.0f;
+            size_t v;
+            for (v = 0; v < DOOM_MUSIC_MAX_VOICES; ++v)
+            {
+                doom_music_voice_t *voice = &g_music_state.voices[v];
+                if (!voice->active)
+                {
+                    continue;
+                }
+                float sample = sinf(voice->phase);
+                voice->phase += voice->phase_step;
+                if (voice->phase > (float)(2.0 * M_PI))
+                {
+                    voice->phase -= (float)(2.0 * M_PI);
+                }
+
+                if (voice->releasing)
+                {
+                    voice->env *= voice->release_factor;
+                    if (voice->env < DOOM_MUSIC_MIN_VOLUME)
+                    {
+                        voice->active = 0;
+                        continue;
+                    }
+                }
+
+                float channel_vol = g_music_state.channel_volume[voice->channel & 0x0F];
+                float pan = g_music_state.channel_pan[voice->channel & 0x0F];
+                if (pan < 0.0f)
+                {
+                    pan = 0.0f;
+                }
+                if (pan > 1.0f)
+                {
+                    pan = 1.0f;
+                }
+
+                float amplitude = sample * voice->env * channel_vol;
+                float left_weight = 1.0f - pan;
+                float right_weight = pan;
+                left += amplitude * left_weight;
+                right += amplitude * right_weight;
+            }
+
+            if (left > 1.0f)
+                left = 1.0f;
+            if (left < -1.0f)
+                left = -1.0f;
+            if (right > 1.0f)
+                right = 1.0f;
+            if (right < -1.0f)
+                right = -1.0f;
+
+            output[out_index++] = (int16_t)(left * 32767.0f);
+            output[out_index++] = (int16_t)(right * 32767.0f);
+            g_music_state.current_sample++;
+        }
+
+        frame_index += frames_to_render;
+    }
+
+    return (int)frame_index;
+}

+void DoomMusic_Init(void)
+{
+    if (g_music_state.initialised)
+    {
+        return;
+    }
+    memset(&g_music_state, 0, sizeof(g_music_state));
+    g_music_state.sample_rate = DOOM_MUSIC_SAMPLE_RATE;
+    g_music_state.samples_per_tick = (double)g_music_state.sample_rate / 140.0;
+    DoomMusic_ResetChannelsAndVoices();
+    wasm_music_init(g_music_state.sample_rate);
+    g_music_state.initialised = 1;
+}
+
+void DoomMusic_Shutdown(void)
+{
+    if (!g_music_state.initialised)
+    {
+        return;
+    }
+    DoomMusic_Stop();
+    free(g_music_state.events);
+    g_music_state.events = NULL;
+    g_music_state.event_count = 0;
+    free(g_music_state.mix_buffer);
+    g_music_state.mix_buffer = NULL;
+    g_music_state.mix_buffer_capacity = 0;
+    wasm_music_shutdown();
+    g_music_state.initialised = 0;
+}
