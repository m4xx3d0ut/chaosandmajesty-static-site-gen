apiVersion: apps/v1
kind: Deployment
metadata:
  name: vllm
  namespace: ai
spec:
  # (make sure selector matches your labels)
  selector:
    matchLabels:
      app.kubernetes.io/name: vllm
  template:
    metadata:
      labels:
        app.kubernetes.io/name: vllm
    spec:
      enableServiceLinks: false
      runtimeClassName: nvidia
      nodeSelector: { accelerator: nvidia }
      containers:
        - name: vllm
          image: vllm/vllm-openai:latest-x86_64
          args: ["--port","8000","--model","zai-org/glm-edge-1.5b-chat","--download-dir","/models","--tensor-parallel-size","1","--dtype","float16"]
          env:
            - name: HUGGING_FACE_HUB_TOKEN
              valueFrom:
                secretKeyRef:
                  name: hf-token
                  key: HUGGING_FACE_HUB_TOKEN
                  optional: true
            - name: VLLM_PORT
              value: "39000"
            # Quadro RTX 8000 (SM 7.5): disable FlashInfer sampler to avoid
            # "too many resources requested for launch" crash at startup.
            - name: VLLM_USE_FLASHINFER_SAMPLER
              value: "0"
            - name: VLLM_USE_FLASHINFER
              value: "0"
          ports:
            - { name: openai, containerPort: 8000 }
          # Probes
          startupProbe:
            httpGet:
              path: /health
              port: openai
            periodSeconds: 5
            timeoutSeconds: 2
            failureThreshold: 60   # ~5 minutes max startup window
          readinessProbe:
            httpGet:
              path: /health
              port: openai
            initialDelaySeconds: 10
            periodSeconds: 5
            timeoutSeconds: 1
            failureThreshold: 3
          livenessProbe:
            httpGet:
              path: /health
              port: openai
            initialDelaySeconds: 60
            periodSeconds: 10
            timeoutSeconds: 2
            failureThreshold: 3
          resources:
            requests: { cpu: "4", memory: "20Gi", nvidia.com/gpu: "1" }
            limits:   { cpu: "8", memory: "40Gi", nvidia.com/gpu: "1" }
          volumeMounts:
            - { name: hf-cache, mountPath: /models }

        - name: cachectl
          image: reg.microk8s.core.home.arpa:32000/ai/cachectl:0.1.0
          env: [{ name: HF_DOWNLOAD_DIR, value: /models }]
          ports: [{ name: cachectl, containerPort: 8081 }]
          volumeMounts: [{ name: hf-cache, mountPath: /models }]
          resources:
            requests: { cpu: "50m", memory: "64Mi" }
            limits:   { cpu: "200m", memory: "256Mi" }

      volumes:
        - name: hf-cache
          persistentVolumeClaim: { claimName: hf-cache }

---
apiVersion: v1
