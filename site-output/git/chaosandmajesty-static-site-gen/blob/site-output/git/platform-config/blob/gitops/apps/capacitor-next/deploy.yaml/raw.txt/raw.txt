# gitops/apps/capacitor-next/deploy.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: capacitor-next
  namespace: flux-system
spec:
  revisionHistoryLimit: 1
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxUnavailable: 1
      maxSurge: 0
  replicas: 1
  selector:
    matchLabels: { app: capacitor-next }
  template:
    metadata:
      labels: { app: capacitor-next }
    spec:
      serviceAccountName: capacitor-next
      restartPolicy: Always
      securityContext:
        runAsNonRoot: true
        seccompProfile: { type: RuntimeDefault }
      containers:
        - name: app
          image: reg.microk8s.core.home.arpa:32000/capacitor-next:next-v0.5.0
          imagePullPolicy: Always
          args:
            - "--port"
            - "3333"
            - "--theme"
            - "dark"
          ports:
            - containerPort: 3333
              name: http
          envFrom:
            - secretRef:
                name: capacitor-next-auth
          resources:
            requests: { cpu: 100m, memory: 128Mi }
            limits:   { cpu: 500m, memory: 384Mi }
          securityContext:
            allowPrivilegeEscalation: false
            runAsNonRoot: true
            capabilities: { drop: ["ALL"] }
          volumeMounts:
            - name: kubeconfig
              mountPath: /home/nonroot/.kube
              readOnly: true
          readinessProbe:
            httpGet: { path: "/", port: 3333 }
            periodSeconds: 10
            timeoutSeconds: 3
          livenessProbe:
            httpGet: { path: "/", port: 3333 }
            periodSeconds: 20
            timeoutSeconds: 3
      volumes:
        - name: kubeconfig
          configMap:
            name: capacitor-next-kubeconfig
