#!/usr/bin/env bash

cd gitops/apps/gitea && \
for f in gitea-init.yaml gitea-inline-config.yaml gitea-postgresql.yaml gitea-runner-token.yaml gitea-valkey.yaml gitea.yaml; do \
  case "$f" in \
    gitea-init.yaml) mv "$f" secret-init.yaml ;; \
    gitea-inline-config.yaml) mv "$f" secret-inline-config.yaml ;; \
    gitea-postgresql.yaml) mv "$f" secret-postgresql.yaml ;; \
    gitea-runner-token.yaml) mv "$f" secret-runner-token.yaml ;; \
    gitea-valkey.yaml) mv "$f" secret-valkey.yaml ;; \
    gitea.yaml) mv "$f" secret-gitea.yaml ;; \
  esac; \
done && \
touch release.yaml
