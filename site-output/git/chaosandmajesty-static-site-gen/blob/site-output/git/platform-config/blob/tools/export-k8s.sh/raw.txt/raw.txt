#!/usr/bin/env bash

set -euo pipefail

mkdir -p .tmp

# Cluster-scoped
for r in $(microk8s kubectl api-resources --verbs=list --namespaced=false -o name); do
  # skip noisy runtime-ish things if present
  case "$r" in
    events.events.k8s.io|events|componentstatuses) continue;;
  esac

  echo "Processing cluster-scoped resource: $r"

  # Use unique filename for each resource
  cluster_file=".tmp/cluster-${r//\//-}.yaml"

  if microk8s kubectl get "$r" -o yaml > "$cluster_file" 2>/dev/null; then
    # Clean the YAML first
    yq eval 'del(.items[].metadata.managedFields) | del(.items[].status)' "$cluster_file" > "${cluster_file}.clean"
    mv "${cluster_file}.clean" "$cluster_file"

    # Check if file has items
    if [[ -s "$cluster_file" ]]; then
      item_count=$(yq eval '.items | length' "$cluster_file" 2>/dev/null || echo "0")
      if [[ "$item_count" -gt 0 ]]; then
        # Convert to JSON once and process with jq
        yq eval -o=json "$cluster_file" | jq -c '.items[]' | while IFS= read -r obj; do
          if [[ -n "$obj" ]]; then
            kind=$(echo "$obj" | jq -r '.kind // "unknown"')
            name=$(echo "$obj" | jq -r '.metadata.name // "unnamed"')
            out="live-export/kubernetes/cluster-scoped/${kind,,}/${name}.yaml"
            mkdir -p "$(dirname "$out")"
            echo "$obj" | jq 'del(.metadata.managedFields,.status,.metadata.resourceVersion,.metadata.uid,.metadata.generation,.metadata.creationTimestamp)' | yq -P > "$out"
            echo "  -> $out"
          fi
        done
      fi
    fi
  else
    echo "  -> Skipped (no access or empty)"
  fi
done

# Namespaced resources
for r in $(microk8s kubectl api-resources --verbs=list --namespaced -o name); do
  case "$r" in
    events.events.k8s.io|events|endpoints|endpointslices.discovery.k8s.io|leases.coordination.k8s.io|controllerrevisions.apps) continue;;
  esac

  echo "Processing namespaced resource: $r"

  for ns in $(microk8s kubectl get ns -o jsonpath='{.items[*].metadata.name}'); do
    ns_file=".tmp/ns-${ns}-${r//\//-}.yaml"

    if microk8s kubectl -n "$ns" get "$r" -o yaml > "$ns_file" 2>/dev/null; then
      # Clean the YAML first
      yq eval 'del(.items[].metadata.managedFields) | del(.items[].status)' "$ns_file" > "${ns_file}.clean"
      mv "${ns_file}.clean" "$ns_file"

      # Check if file has items
      if [[ -s "$ns_file" ]]; then
        item_count=$(yq eval '.items | length' "$ns_file" 2>/dev/null || echo "0")
        if [[ "$item_count" -gt 0 ]]; then
          # Convert to JSON once and process with jq
          yq eval -o=json "$ns_file" | jq -c '.items[]' | while IFS= read -r obj; do
            if [[ -n "$obj" ]]; then
              kind=$(echo "$obj" | jq -r '.kind // "unknown"')
              name=$(echo "$obj" | jq -r '.metadata.name // "unnamed"')
              # Skip auto-generated SA token secrets
              if [[ "$kind" == "Secret" ]]; then
                stype=$(echo "$obj" | jq -r '.type // ""')
                [[ "$stype" == "kubernetes.io/service-account-token" ]] && continue
              fi
              out="live-export/kubernetes/namespaces/$ns/${kind,,}/${name}.yaml"
              mkdir -p "$(dirname "$out")"
              echo "$obj" | jq 'del(.metadata.managedFields,.status,.metadata.resourceVersion,.metadata.uid,.metadata.generation,.metadata.creationTimestamp,.metadata.annotations."kubectl.kubernetes.io/last-applied-configuration")' | yq -P > "$out"
              echo "    -> $out"
            fi
          done
        fi
      fi
    fi
  done
done

echo "Export complete!"
