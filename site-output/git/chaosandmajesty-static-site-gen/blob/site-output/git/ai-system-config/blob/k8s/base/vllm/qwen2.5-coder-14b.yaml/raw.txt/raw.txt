# k8s/base/vllm/qwen2.5-coder-14b.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: vllm-qwen2p5-coder-14b
  namespace: ai
  labels: { app.kubernetes.io/name: vllm-qwen2p5-coder-14b, app.kubernetes.io/part-of: ai-system, ai.zlabs.dev/component: vllm }
  annotations:
    kustomize.toolkit.fluxcd.io/ignore: "true"
    ai.zlabs.dev/maxReplicas: "1"
spec:
  replicas: 0
  selector:
    matchLabels: { app.kubernetes.io/name: vllm-qwen2p5-coder-14b, app.kubernetes.io/part-of: ai-system }
  template:
    metadata:
      labels: { app.kubernetes.io/name: vllm-qwen2p5-coder-14b, app.kubernetes.io/part-of: ai-system, ai.zlabs.dev/component: vllm }
    spec:
      enableServiceLinks: false
      runtimeClassName: nvidia
      nodeSelector: { accelerator: nvidia }
      terminationGracePeriodSeconds: 1200
      containers:
        - name: vllm
          image: vllm/vllm-openai:latest-x86_64
          args:
            - "--model=Qwen/Qwen2.5-Coder-14B-Instruct"
            - "--port=8000"
            - "--trust-remote-code"
            - "--max-model-len=4096"
            - "--gpu-memory-utilization=0.65"
            - "--swap-space=8"
            - "--enforce-eager"
            - "--disable-custom-all-reduce"
            - "--enable-log-requests=false"   # was --disable-log-requests
          env:
            - name: HUGGING_FACE_HUB_TOKEN
              valueFrom: { secretKeyRef: { name: hf-token, key: HUGGING_FACE_HUB_TOKEN, optional: true } }
            # Turing-safe: avoid FlashInfer entirely on RTX 8000
            - { name: VLLM_USE_FLASHINFER, value: "0" }
            - { name: VLLM_USE_FLASHINFER_SAMPLER, value: "0" }
          ports: [ { name: http, containerPort: 8000 }, { name: cachectl, containerPort: 8081 } ]

          # <-- NEW: long startup window while model loads
          startupProbe:
            httpGet: { path: /health, port: http }
            periodSeconds: 5
            timeoutSeconds: 2
            failureThreshold: 240   # ~20 minutes max warmup window

          # Ready only after OpenAI API is up
          readinessProbe:
            httpGet: { path: /v1/models, port: http }
            initialDelaySeconds: 5
            periodSeconds: 5
            timeoutSeconds: 2
            failureThreshold: 120   # tolerate long first answer

          # Liveness should be very forgiving
          livenessProbe:
            httpGet: { path: /health, port: http }
            initialDelaySeconds: 1200
            periodSeconds: 15
            timeoutSeconds: 3
            failureThreshold: 3

          resources:
            requests: { cpu: "8", memory: "32Gi", nvidia.com/gpu: "1" }
            limits:   { cpu: "16", memory: "96Gi", nvidia.com/gpu: "1" }
          volumeMounts: [ { name: hf-cache, mountPath: /root/.cache/huggingface } ]
      volumes: [ { name: hf-cache, persistentVolumeClaim: { claimName: hf-cache } } ]
---
apiVersion: v1
kind: Service
metadata:
  name: vllm-qwen2p5-coder-14b
  namespace: ai
  labels: { app.kubernetes.io/name: vllm-qwen2p5-coder-14b }
spec:
  selector: { app.kubernetes.io/name: vllm-qwen2p5-coder-14b }
  ports:
    - { name: openai, port: 8000, targetPort: 8000 }
    - { name: cachectl, port: 8081, targetPort: 8081 }
