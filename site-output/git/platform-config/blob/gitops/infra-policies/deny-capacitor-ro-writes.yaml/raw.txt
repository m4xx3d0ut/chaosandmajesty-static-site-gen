apiVersion: admissionregistration.k8s.io/v1
kind: ValidatingAdmissionPolicy
metadata:
  name: deny-capacitor-ro-writes
spec:
  failurePolicy: Fail
  matchConstraints:
    resourceRules:
      - apiGroups: ["", "apps", "batch"]
        apiVersions: ["*"]
        operations: ["*"]
        resources:
          - pods
          - pods/exec
          - pods/attach
          - pods/portforward
          - pods/eviction
          - pods/ephemeralcontainers
          - deployments
          - deployments/scale
          - replicasets
          - replicasets/scale
          - statefulsets
          - statefulsets/scale
          - daemonsets
          - jobs
          - cronjobs
  validations:
    - expression: request.userInfo.username != "system:serviceaccount:flux-system:capacitor-next-ro"
      message: "capacitor-next-ro is view-only; write-like operations are blocked by policy."
