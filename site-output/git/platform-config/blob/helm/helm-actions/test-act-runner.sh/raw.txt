#!/usr/bin/env sh

kubectl exec -n git-local -c act-runner gitea-actions-act-runner-0 -- sh -lc '
DOCKER_HOST=tcp://127.0.0.1:2376 DOCKER_TLS_VERIFY=1 DOCKER_CERT_PATH=/certs/client \
docker run --rm \
  -e DOCKER_HOST=tcp://docker:2376 \
  -e DOCKER_TLS_VERIFY=1 \
  -e DOCKER_CERT_PATH=/certs/client \
  -v /certs/client:/certs/client:ro \
  -v /usr/local/share/ca-certificates/gitea-root.crt:/usr/local/share/ca-certificates/gitea-root.crt:ro \
  --add-host=docker:host-gateway \
  alpine:3.20 sh -lc "
    apk add --no-cache ca-certificates docker-cli >/dev/null &&
    update-ca-certificates >/dev/null &&
    docker version
  "
'
