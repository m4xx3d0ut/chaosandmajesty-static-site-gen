#!/usr/bin/env bash

set -euo pipefail
for ns in $(microk8s helm3 list -A | awk 'NR>1{print $2}' | sort -u); do
  for rel in $(microk8s helm3 -n "$ns" list | awk 'NR>1{print $1}'); do
    base="live-export/helm/$ns/$rel"
    mkdir -p "$base"
    microk8s helm3 -n "$ns" get values "$rel" > "$base/values.yaml" || true
    microk8s helm3 -n "$ns" get values "$rel" --all > "$base/values-all.yaml" || true
    microk8s helm3 -n "$ns" get manifest "$rel" > "$base/manifest.yaml" || true
  done
done
