  1. #!/usr/bin/env bash
  2. ​
  3. # Strict mode
  4. set -euo pipefail
  5. ​
  6. # Installs stagit from source into /usr/local.
  7. # Requires sudo privileges (GitHub-hosted and microk8s runners allow this).
  8. ​
  9. SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
  10. ​
  11. STAGIT_REF="master"
  12. DEFAULT_REMOTES=(
  13. "git://git.codemadness.org/stagit"
  14. "git://git.2f30.org/stagit"
  15. "https://github.com/rski/stagit.git"
  16. )
  17. ​
  18. if [[ -n "${STAGIT_REMOTE_OVERRIDE:-}" ]]; then
  19. DEFAULT_REMOTES=("${STAGIT_REMOTE_OVERRIDE}" "${DEFAULT_REMOTES[@]}")
  20. fi
  21. ​
  22. if command -v stagit >/dev/null 2>&1 && command -v stagit-index >/dev/null 2>&1; then
  23. exit 0
  24. fi
  25. ​
  26. echo "[install-stagit] Updating apt cache and installing build prerequisites" >&2
  27. sudo apt-get update -y
  28. sudo apt-get install -y --no-install-recommends build-essential pkg-config libgit2-dev git ca-certificates curl
  29. ​
  30. tmpdir="$(mktemp -d)"
  31. trap 'rm -rf "${tmpdir}"' EXIT
  32. ​
  33. srcdir=""
  34. export GIT_TERMINAL_PROMPT=0
  35. ​
  36. for remote in "${DEFAULT_REMOTES[@]}"; do
  37. [[ -z "$remote" ]] && continue
  38. echo "[install-stagit] Cloning ${remote} (${STAGIT_REF})" >&2
  39. rm -rf "${tmpdir}/stagit"
  40. if [[ "$remote" == git://* ]]; then
  41. if git clone --depth 1 "$remote" "${tmpdir}/stagit"; then
  42. srcdir="${tmpdir}/stagit"
  43. break
  44. fi
  45. else
  46. if git clone --depth 1 --branch "${STAGIT_REF}" "$remote" "${tmpdir}/stagit"; then
  47. srcdir="${tmpdir}/stagit"
  48. break
  49. fi
  50. echo "[install-stagit] Branch clone failed for ${remote}; attempting default branch" >&2
  51. rm -rf "${tmpdir}/stagit"
  52. if git clone --depth 1 "$remote" "${tmpdir}/stagit"; then
  53. srcdir="${tmpdir}/stagit"
  54. break
  55. fi
  56. fi
  57. echo "[install-stagit] Clone failed for ${remote}; trying next mirror" >&2
  58. done
  59. ​
  60. if [[ -z "$srcdir" ]]; then
  61. for local_tar in "${SCRIPT_DIR}/stagit"/*.tar.gz; do
  62. [[ ! -f "$local_tar" ]] && continue
  63. echo "[install-stagit] Using bundled tarball ${local_tar}" >&2
  64. cp "$local_tar" "${tmpdir}/stagit.tar.gz"
  65. tar -xzf "${tmpdir}/stagit.tar.gz" -C "$tmpdir"
  66. srcdir="$(find "$tmpdir" -maxdepth 1 -type d -name 'stagit-*' -print -quit)"
  67. if [[ -n "$srcdir" ]]; then
  68. break
  69. fi
  70. done
  71. fi
  72. ​
  73. if [[ -z "$srcdir" ]]; then
  74. echo "[install-stagit] Falling back to tarball download" >&2
  75. tarball_urls=(
  76. "https://git.codemadness.org/stagit/snapshot/stagit-${STAGIT_REF}.tar.gz"
  77. "https://git.2f30.org/stagit/snapshot/stagit-${STAGIT_REF}.tar.gz"
  78. "https://codeload.github.com/rski/stagit/tar.gz/refs/heads/${STAGIT_REF}"
  79. )
  80. for url in "${tarball_urls[@]}"; do
  81. [[ -z "$url" ]] && continue
  82. if curl -fsSL "$url" -o "${tmpdir}/stagit.tar.gz"; then
  83. tar -xzf "${tmpdir}/stagit.tar.gz" -C "$tmpdir"
  84. srcdir="$(find "$tmpdir" -maxdepth 1 -type d -name 'stagit-*' -print -quit)"
  85. if [[ -n "$srcdir" ]]; then
  86. break
  87. fi
  88. fi
  89. done
  90. fi
  91. ​
  92. if [[ -z "$srcdir" ]]; then
  93. echo "[install-stagit] Unable to obtain stagit sources from any mirror or local cache" >&2
  94. exit 1
  95. fi
  96. ​
  97. cd "$srcdir"
  98. ​
  99. echo "[install-stagit] Building stagit" >&2
  100. make
  101. ​
  102. echo "[install-stagit] Installing to /usr/local/bin" >&2
  103. sudo make install
  104. ​
  105. echo "[install-stagit] stagit installed at $(command -v stagit)" >&2
