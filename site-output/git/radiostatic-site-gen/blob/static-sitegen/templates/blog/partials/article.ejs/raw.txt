<%
  const resolveAsset = (typeof buildAssetHref === 'function')
    ? buildAssetHref
    : function(asset) { return asset; };
  const resolveInternal = (typeof buildInternalHref === 'function')
    ? buildInternalHref
    : function(href) { return href; };
  const adjacent = post.adjacent || {};
  const navPrev = adjacent.previous || null;
  const navNext = adjacent.next || null;
  const navFlagFromPost = (post && typeof post.showArticleNav === 'boolean') ? post.showArticleNav : null;
  const explicitNavFlag = typeof showArticleNav === 'boolean' ? showArticleNav : navFlagFromPost;
  const shouldRenderNav = explicitNavFlag !== null ? explicitNavFlag : !post.isHidden;
  const prevHrefValue = navPrev ? (navPrev.relativeHref || navPrev.publicHref || '') : '';
  const nextHrefValue = navNext ? (navNext.relativeHref || navNext.publicHref || '') : '';
  const prevHref = prevHrefValue ? resolveInternal(prevHrefValue) : '#';
  const nextHref = nextHrefValue ? resolveInternal(nextHrefValue) : '#';
  const prevFragmentValue = navPrev ? (navPrev.fragmentPath || navPrev.fragmentHref || '') : '';
  const nextFragmentValue = navNext ? (navNext.fragmentPath || navNext.fragmentHref || '') : '';
  const prevFragmentHref = prevFragmentValue ? resolveInternal(prevFragmentValue) : '';
  const nextFragmentHref = nextFragmentValue ? resolveInternal(nextFragmentValue) : '';
%>

<article class="blog-article-body" data-slug="<%= post.slug %>">
  <header class="blog-article-header">
    <h1 class="blog-article-title"><%= post.title %></h1>
    <p class="blog-article-meta">
      <% if (post.displayPublishedAt) { %>
        <span><%= post.displayPublishedAt %></span>
      <% } %>
      <% if (post.readingMinutes) { %>
        <span aria-hidden="true">•</span>
        <span><%= post.readingMinutes %> min read</span>
      <% } %>
      <% if (post.author) { %>
        <span aria-hidden="true">•</span>
        <span><%= post.author.name %></span>
      <% } %>
    </p>
  </header>

  <% if (post.heroImagePath) { %>
    <figure class="blog-hero">
      <img src="<%= resolveAsset(post.heroImagePath) %>" alt="<%= post.title %> hero image">
    </figure>
  <% } %>

  <div class="blog-article-content">
    <%- post.html %>
  </div>

  <% if (post.author) { %>
    <footer class="blog-author-card">
      <% if (post.author.avatarPath) { %>
        <img class="blog-author-avatar" src="<%= resolveAsset(post.author.avatarPath) %>" alt="<%= post.author.name %> avatar">
      <% } %>
      <div class="blog-author-details">
        <strong class="blog-author-name"><%= post.author.name %></strong>
        <% if (post.author.title) { %>
          <div class="blog-author-title"><%= post.author.title %></div>
        <% } %>
        <% if (post.author.bio) { %>
          <p class="blog-author-bio"><%= post.author.bio %></p>
        <% } %>
        <% if (post.author.social && post.author.social.length) { %>
          <ul class="blog-author-social">
            <% post.author.social.forEach(function(link) { %>
              <li><a href="<%= link.href %>" target="_blank" rel="noopener"><%= link.label %></a></li>
            <% }); %>
          </ul>
        <% } %>
      </div>
    </footer>
  <% } %>

  <% if (shouldRenderNav) { %>
    <nav class="blog-article-nav" aria-label="Post navigation">
      <a
        class="blog-article-nav-link<%= navPrev ? '' : ' is-disabled' %>"
        data-blog-nav="previous"
        <%= navPrev ? '' : 'aria-disabled="true" tabindex="-1"' %>
        href="<%= navPrev ? prevHref : '#' %>"
        data-default-href="<%= navPrev ? prevHref : '' %>"
        data-default-fragment="<%= navPrev ? prevFragmentHref : '' %>"
        data-fragment-href="<%= navPrev ? prevFragmentHref : '' %>"
        data-target-slug="<%= navPrev ? navPrev.slug : '' %>"
        data-post-slug="<%= navPrev ? navPrev.slug : '' %>"
      >
        <span class="blog-article-nav-meta">Previous</span>
        <span class="blog-article-nav-title"><%= navPrev ? navPrev.title : 'No newer posts' %></span>
      </a>
      <span class="blog-article-nav-delimiter" aria-hidden="true">&nbsp;•&nbsp;</span>
      <a
        class="blog-article-nav-link<%= navNext ? '' : ' is-disabled' %>"
        data-blog-nav="next"
        <%= navNext ? '' : 'aria-disabled="true" tabindex="-1"' %>
        href="<%= navNext ? nextHref : '#' %>"
        data-default-href="<%= navNext ? nextHref : '' %>"
        data-default-fragment="<%= navNext ? nextFragmentHref : '' %>"
        data-fragment-href="<%= navNext ? nextFragmentHref : '' %>"
        data-target-slug="<%= navNext ? navNext.slug : '' %>"
        data-post-slug="<%= navNext ? navNext.slug : '' %>"
      >
        <span class="blog-article-nav-meta">Next</span>
        <span class="blog-article-nav-title"><%= navNext ? navNext.title : 'No older posts' %></span>
      </a>
    </nav>
  <% } %>
</article>
