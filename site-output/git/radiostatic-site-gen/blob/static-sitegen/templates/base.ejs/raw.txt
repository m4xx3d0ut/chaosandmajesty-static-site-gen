  <% 
    const isExternalHref = (href) => /^(?:[a-z][a-z0-9+.-]*:|\/\/)/i.test(href || '');
    const normalizeStyle = (href) => {
      if (!href) {
        return '';
      }
      if (isExternalHref(href)) {
        return href;
      }
      if (/^(?:\.{1,2}\/)/.test(href)) {
        return href;
      }
      if (href.startsWith('/')) {
        return href;
      }
      const depthValue = Number.isInteger(page.baseDepth) && page.baseDepth >= 0
        ? page.baseDepth
        : (isSection ? 1 : 0);
      const depthPrefix = depthValue > 0 ? '../'.repeat(depthValue) : '';
      return depthPrefix + href.replace(/^\/+/, '');
    };
    const collectedStyles = [];
    if (Array.isArray(site.styles) && site.styles.length > 0) {
      collectedStyles.push(...site.styles);
    }
    if (Array.isArray(page.styles) && page.styles.length > 0) {
      collectedStyles.push(...page.styles);
    }
    if (collectedStyles.length === 0) {
      collectedStyles.push('assets/modern.css');
    }
    const seenStyles = new Set();
    const normalizedStyles = collectedStyles
      .map(normalizeStyle)
      .filter((href) => {
        if (!href) {
          return false;
        }
        if (seenStyles.has(href)) {
          return false;
        }
        seenStyles.add(href);
        return true;
      });
  %>
  <% normalizedStyles.forEach(function(styleHref) { %>
    <link rel="stylesheet" href="<%= styleHref %>">
  <% }); %>

    (function(seed) {
      var existingSeed = window.radiostaticConsoleDispatches || window.sitegenConsoleDispatches || window.cmConsoleDispatches;
      var existing = Array.isArray(existingSeed) ? existingSeed.slice() : [];
      var seen = new Set();
      existing.forEach(function(entry) {
        if (entry && entry.slug) {
          seen.add(entry.slug);
        }
      });
      if (Array.isArray(seed)) {
        seed.forEach(function(entry) {
          if (!entry || (entry.slug && seen.has(entry.slug))) {
            return;
          }
          existing.push(entry);
          if (entry.slug) {
            seen.add(entry.slug);
          }
        });
      }
      window.radiostaticConsoleDispatches = existing;
      window.sitegenConsoleDispatches = existing;
      window.cmConsoleDispatches = existing;
    })(<%- JSON.stringify(dispatchData) %>);

    const doomBaseDepthValue = (typeof page.baseDepth === 'number' && page.baseDepth > 0) ? '../'.repeat(page.baseDepth) : '';
