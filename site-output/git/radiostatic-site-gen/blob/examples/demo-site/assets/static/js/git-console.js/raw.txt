(function() {
  const ROOT_SELECTOR = '.git-grid';

  function qs(root, selector) {
    return root.querySelector(selector);
  }

  function qsa(root, selector) {
    return Array.from(root.querySelectorAll(selector));
  }

  function normalizeRepos(data) {
    if (!Array.isArray(data)) {
      return [];
    }
    return data.map(function(entry) {
      return {
        id: entry.id || entry.slug || entry.label || 'repo',
        label: entry.label || entry.name || entry.id || 'Repository',
        stage: entry.stage || '',
        description: entry.description || '',
        default: Boolean(entry.default),
        log: entry.log || '',
        files: entry.files || '',
        readme: entry.readme || '',
        refs: entry.refs || '',
        cloneCommand: entry.cloneCommand || ''
      };
    }).filter(function(entry) {
      return entry.log;
    });
  }

  function buildRepoList(root, repos, selectRepo) {
    const list = qs(root, '[data-git-list]');
    if (!list) {
      return;
    }
    list.innerHTML = '';
    repos.forEach(function(repo) {
      const li = document.createElement('li');
      li.className = 'git-repo-list-item';
      const button = document.createElement('button');
      button.type = 'button';
      button.className = 'git-repo-button';
      button.dataset.repoId = repo.id;
      button.innerHTML = `
        <span class="git-repo-label">${repo.label}</span>
        <span class="git-repo-meta">${repo.stage ? repo.stage + ' · ' : ''}${repo.description}</span>
      `;
      button.addEventListener('click', function() {
        selectRepo(repo.id);
      });
      li.appendChild(button);
      list.appendChild(li);
    });
  }

  function updateActiveButton(root, repoId) {
    qsa(root, '.git-repo-button').forEach(function(button) {
      button.classList.toggle('is-active', button.dataset.repoId === repoId);
    });
  }

  function updateActiveMeta(root, repo) {
    const label = qs(root, '[data-git-active-label]');
    const stage = qs(root, '[data-git-active-stage]');
    if (label) {
      label.textContent = repo.label;
    }
    if (stage) {
      stage.textContent = repo.stage ? `${repo.stage} · ${repo.description || ''}`.trim() : repo.description;
    }
  }

  function configureTabs(root, repo) {
    qsa(root, '.git-tab-button').forEach(function(button) {
      const tab = button.dataset.gitTab;
      let target = '';
      if (tab === 'log') target = repo.log;
      else if (tab === 'files') target = repo.files;
      else if (tab === 'readme') target = repo.readme;
      else if (tab === 'refs') target = repo.refs;

      if (target) {
        button.disabled = false;
        button.setAttribute('hx-get', target);
      } else {
        button.disabled = true;
        button.removeAttribute('hx-get');
      }
    });
  }

  function markActiveTab(root, tab) {
    qsa(root, '.git-tab-button').forEach(function(button) {
      button.classList.toggle('is-active', button.dataset.gitTab === tab);
    });
  }

  function wireTabs(root, loadTab) {
    qsa(root, '.git-tab-button').forEach(function(button) {
      button.addEventListener('click', function() {
        if (!button.disabled) {
          loadTab(button.dataset.gitTab);
        }
      });
    });
  }

  function loadFragment(root, repo, tab) {
    const targetButton = root.querySelector(`.git-tab-button[data-git-tab="${tab}"]`);
    if (!targetButton || !targetButton.getAttribute('hx-get')) {
      return;
    }
    root.dataset.gitView = tab;
    if (tab === 'files') {
      root.classList.add('is-files-view');
    } else {
      root.classList.remove('is-files-view');
    }
    if (window.htmx) {
      window.htmx.ajax('GET', targetButton.getAttribute('hx-get'), {
        target: '#git-panel',
        swap: 'innerHTML'
      });
    }
    markActiveTab(root, tab);
  }

  function wireFilter(root) {
    const input = qs(root, '[data-git-filter]');
    const list = qs(root, '[data-git-list]');
    if (!input || !list) return;
    input.addEventListener('input', function() {
      const value = input.value.trim().toLowerCase();
      qsa(list, '.git-repo-list-item').forEach(function(item) {
        const label = item.textContent.toLowerCase();
        item.hidden = value && !label.includes(value);
      });
    });
  }

  function wireClipboard(root) {
    qsa(root, '[data-git-copy]').forEach(function(button) {
      if (button.dataset.gitBound === 'true') {
        return;
      }
      button.dataset.gitBound = 'true';
      button.addEventListener('click', function() {
        const container = button.closest('[data-git-command]') || button.closest('.git-panel-clone');
        let commandText = '';
        if (container) {
          const commandEl = container.querySelector('[data-git-command]') || container.querySelector('[data-git-copy-target]') || container.querySelector('[data-git-command]');
          if (commandEl && commandEl.textContent) {
            commandText = commandEl.textContent.trim();
          }
          if (!commandText) {
            const code = container.querySelector('.git-clone-command');
            commandText = code ? code.textContent.trim() : '';
          }
        }
        if (!commandText) {
          const code = button.previousElementSibling;
          commandText = code ? code.textContent.trim() : '';
        }
        if (!commandText) {
          return;
        }
        navigator.clipboard.writeText(commandText).then(function() {
          button.classList.add('is-copied');
          setTimeout(function() {
            button.classList.remove('is-copied');
          }, 1800);
        }).catch(function() {
          console.warn('Unable to copy command');
        });
      });
    });
  }

  function wireFilePreview(root) {
    const treeButtons = qsa(root, '.git-tree-entry');
    if (treeButtons.length === 0) {
      return;
    }
    const preview = root.querySelector('[data-git-file-preview]');
    treeButtons.forEach(function(button) {
      if (button.dataset.gitBound === 'true') {
        return;
      }
      button.dataset.gitBound = 'true';
      button.addEventListener('click', function() {
        treeButtons.forEach(function(item) {
          item.classList.toggle('is-active', item === button);
        });
        if (preview) {
          const fileName = button.dataset.gitFile || button.textContent.trim();
          const title = preview.querySelector('.git-file-title');
          if (title) {
            title.textContent = fileName;
          }
        }
      });
    });
  }

  function hydrateFragments() {
    const panel = qs(document, '#git-panel');
    if (!panel) {
      return;
    }
    wireClipboard(panel);
    wireFilePreview(panel);
  }

  document.addEventListener('DOMContentLoaded', function() {
    const root = document.querySelector(ROOT_SELECTOR);
    if (!root) {
      return;
    }

    const source = root.getAttribute('data-git-source');
    let repos = [];

    function selectRepo(repoId, initialTab) {
      const repo = repos.find(function(candidate) { return candidate.id === repoId; });
      if (!repo) {
        return;
      }
      updateActiveButton(root, repo.id);
      updateActiveMeta(root, repo);
      configureTabs(root, repo);
      loadFragment(root, repo, initialTab || 'log');
      root.dataset.activeRepo = repo.id;
    }

    wireTabs(root, function(tab) {
      const activeId = root.dataset.activeRepo;
      const repo = repos.find(function(candidate) { return candidate.id === activeId; });
      if (!repo) {
        return;
      }
      loadFragment(root, repo, tab);
    });
    wireFilter(root);

    if (source) {
      fetch(source)
        .then(function(resp) { return resp.json(); })
        .then(function(json) {
          repos = normalizeRepos(json);
          if (repos.length === 0) {
            const list = qs(root, '[data-git-list]');
            if (list) {
              list.innerHTML = '<li class="git-empty-state">No repositories available.</li>';
            }
            return;
          }
          buildRepoList(root, repos, selectRepo);
          const defaultRepo = repos.find(function(repo) { return repo.default; }) || repos[0];
          selectRepo(defaultRepo.id, 'log');
        })
        .catch(function(error) {
          console.error('Unable to load git repos', error);
          const panel = qs(document, '#git-panel');
          if (panel) {
            panel.innerHTML = '<p class="git-panel-error">Unable to load repository data. Ensure the fragments were generated during the build.</p>';
          }
        });
    }
  });

  document.addEventListener('htmx:afterSwap', function(event) {
    if (!event || !event.target || event.target.id !== 'git-panel') {
      return;
    }
    hydrateFragments();
  });
})();
