#!/bin/bash
# prepare-demo-assets.sh
# Copies example assets into static-sitegen/assets for local development.
# Usage: ./prepare-demo-assets.sh [source-dir]
# Default source directory is 'examples/demo-site/assets'.

set -euo pipefail

SOURCE_DIR="${1:-examples/demo-site/assets}"
TARGET_DIR="static-sitegen/assets"

if [ ! -d "$SOURCE_DIR" ]; then
  echo "Error: source directory '$SOURCE_DIR' not found" >&2
  exit 1
fi

echo "Syncing assets from '$SOURCE_DIR' into '$TARGET_DIR'..."

mkdir -p "$TARGET_DIR"
if command -v rsync >/dev/null 2>&1; then
  rsync -av --delete "$SOURCE_DIR"/ "$TARGET_DIR"/
else
  echo "rsync not found; falling back to cp" >&2
  rm -rf "$TARGET_DIR"/*
  cp -R "$SOURCE_DIR"/. "$TARGET_DIR"/
fi

echo "Done. Assets are ready for local builds."
