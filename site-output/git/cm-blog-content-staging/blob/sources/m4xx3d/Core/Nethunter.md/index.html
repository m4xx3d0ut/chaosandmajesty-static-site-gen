SD card bind mount for Kali RootFS

su --mount-master -c "mount --bind /storage/emulated/0/nhsys/ /data/local/nhsystem/"

Magisk Boot Script to create bind mount and output to logcat.

shieldtablet:/ $ su
shieldtablet:/ # cat /data/adb/service.d/nhbind.sh
#!/system/bin/sh
{
# Bind mount ext sdcard nhsys to internal nhsystem
su --mount-master -c "mount --bind /storage/emulated/0/nhsys/ /data/local/nhsystem/"
} | tee /dev/tty | while read line; do
 log -p v -t "$0" "$line"
done

Metasploit Postgres connection to Termux Postgresql, avoids issues with SYSVIPC SHMMEM in Kali chroot.

pkg update && pkg install openssl-tool postgresql
mkdir {/path/to/db/dir}
pg_ctl initdb -D {/path/to/db/dir}
pg_ctl -D {/path/to/db/dir} start
pg_ctl -D {/path/to/db/dir} stop
createuser --superuser --pwprompt msf
pw: LmnKsFF962a1

# psql shell
psql postgres
CREATE DATABASE msf;
\l
\q

# kali shell
msfconsole
db_connect msf:LmnKsFF962a1@127.0.0.1/msf
