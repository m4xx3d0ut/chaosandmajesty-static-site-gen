#!/usr/bin/env python3
"""Synchronize blog content into the Chaos & Majesty site generator tree."""
from __future__ import annotations

import argparse
import shutil
from pathlib import Path

import yaml


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Copy blog markdown and assets into the CM site generator repo",
    )
    parser.add_argument(
        "source",
        help="Path to the posts repository (containing blog/).",
    )
    parser.add_argument(
        "target",
        help="Path to the Chaos & Majesty site generator repository root.",
    )
    return parser.parse_args()


def load_front_matter(path: Path) -> tuple[dict, str]:
    text = path.read_text(encoding="utf-8")
    if not text.startswith("---\n"):
        return {}, text
    _, header, remainder = text.split("---\n", 2)
    data = yaml.safe_load(header) or {}
    return data, remainder


def write_front_matter(path: Path, data: dict, remainder: str) -> None:
    header = yaml.safe_dump(data, sort_keys=False).strip()
    content = f"---\n{header}\n---\n{remainder.lstrip()}"
    path.write_text(content, encoding="utf-8")


def normalise_hero_image(data: dict, slug: str) -> dict:
    hero = data.get("heroImage")
    if not hero:
        return data
    filename = Path(hero).name
    data["heroImage"] = f"assets/static/img/blog/{slug}/{filename}"
    return data


def copy_markdown(source_blog: Path, target_blog: Path) -> list[tuple[str, Path]]:
    target_blog.mkdir(parents=True, exist_ok=True)
    copied: list[tuple[str, Path]] = []
    for md_path in sorted(source_blog.glob("*.md")):
        dest = target_blog / md_path.name
        shutil.copy2(md_path, dest)
        data, remainder = load_front_matter(dest)
        slug = data.get("slug") or md_path.stem
        data = normalise_hero_image(data, slug)
        write_front_matter(dest, data, remainder)
        copied.append((slug, dest))
        print(f"Copied {md_path.name} -> {dest}")
    return copied


def copy_assets(source_assets: Path, target_assets_list: list[Path]) -> None:
    if not source_assets.exists():
        print("No assets directory found; skipping asset sync.")
        return
    slugs = [p for p in source_assets.iterdir() if p.is_dir()]
    for slug_dir in slugs:
        slug = slug_dir.name
        for dest_root in target_assets_list:
            dest = dest_root / slug
            dest.parent.mkdir(parents=True, exist_ok=True)
            shutil.copytree(slug_dir, dest, dirs_exist_ok=True)
        print(f"Synced assets for {slug}")


def main() -> None:
    args = parse_args()
    source_root = Path(args.source).resolve()
    target_root = Path(args.target).resolve()

    source_blog = source_root / "blog"
    source_assets = source_blog / "assets"
    target_blog = target_root / "content" / "blog"
    target_assets = target_root / "static-sitegen" / "assets" / "static" / "img" / "blog"
    cm_source_assets = target_root / "cm-source" / "static" / "img" / "blog"

    if not source_blog.exists():
        raise SystemExit(f"Blog directory not found at {source_blog}")
    if not target_blog.parent.exists():
        raise SystemExit(f"Target repo missing content/ directory at {target_blog.parent}")

    print(f"Source blog: {source_blog}")
    print(f"Target blog: {target_blog}")

    copy_markdown(source_blog, target_blog)
    copy_assets(source_assets, [target_assets, cm_source_assets])
    print("Sync complete. Run the site generator build to validate the import.")


if __name__ == "__main__":
    main()
