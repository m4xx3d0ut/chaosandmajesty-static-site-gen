#!/usr/bin/env python3
"""Convert markdown drafts under content/ into blog-ready posts with assets."""
from __future__ import annotations

import argparse
import datetime as dt
import math
import re
import shutil
from pathlib import Path
from typing import Dict, List, Tuple

import yaml

DEFAULT_AUTHOR = "m4xx3d0ut"
WORDS_PER_MINUTE = 200
IMAGE_PATTERN = re.compile(r"!\[[^\]]*\]\(([^)]+)\)")


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--content-dir",
        type=Path,
        default=Path("content"),
        help="Directory containing draft markdown files.",
    )
    parser.add_argument(
        "--output-dir",
        type=Path,
        default=Path("blog"),
        help="Destination directory for blog-ready posts.",
    )
    parser.add_argument(
        "--author",
        type=str,
        default=DEFAULT_AUTHOR,
        help="Default author to assign when missing.",
    )
    parser.add_argument(
        "--force",
        action="store_true",
        help="Overwrite existing output files even if they look up-to-date.",
    )
    return parser.parse_args()


def split_front_matter(raw: str) -> Tuple[Dict[str, object], str]:
    if not raw.startswith("---\n"):
        return {}, raw
    parts = raw.split("\n---\n", 1)
    if len(parts) != 2:
        return {}, raw
    header_raw, body = parts
    header = yaml.safe_load(header_raw[4:]) or {}
    return header, body


def normalise_date(value: str | None) -> str:
    if not value:
        return dt.date.today().isoformat()
    value = value.strip()
    for suffix in ("Z", "z"):
        if value.endswith(suffix):
            value = value[:-1]
            break
    try:
        return dt.datetime.fromisoformat(value).date().isoformat()
    except ValueError:
        try:
            return dt.datetime.strptime(value, "%Y-%m-%d").date().isoformat()
        except ValueError:
            return dt.date.today().isoformat()


def slugify(text: str) -> str:
    cleaned = re.sub(r"[^a-zA-Z0-9]+", "-", text.lower()).strip("-")
    return cleaned or "post"


def summarise(markdown_body: str, limit: int = 220) -> str:
    paragraphs = [p.strip() for p in markdown_body.split("\n\n") if p.strip()]
    for paragraph in paragraphs:
        if paragraph.startswith("#"):
            continue
        plain = re.sub(r"\s+", " ", paragraph)
        if len(plain) <= limit:
            return plain
        return plain[: limit - 1].rstrip() + "…"
    return ""


def estimate_reading_minutes(text: str) -> int:
    words = re.findall(r"\b\w+\b", text)
    return max(1, math.ceil(len(words) / WORDS_PER_MINUTE))


def collect_tags(source_path: Path, content_root: Path, metadata: Dict[str, object]) -> List[str]:
    existing = metadata.get("tags")
    if isinstance(existing, list):
        return [str(tag) for tag in existing]
    try:
        relative = source_path.relative_to(content_root)
    except ValueError:
        relative = source_path.name
    tags = [slugify(part).replace("-", " ") for part in Path(relative).parts[:-1]]
    return sorted({tag for tag in tags if tag})


def rewrite_images(body: str, source: Path, asset_root: Path, slug: str) -> Tuple[str, str | None]:
    hero = None
    target_dir = asset_root / slug
    replacements: Dict[str, str] = {}
    for match in IMAGE_PATTERN.finditer(body):
        raw_path = match.group(1).strip()
        if raw_path.startswith("http://") or raw_path.startswith("https://"):
            continue
        source_path = (source.parent / raw_path).resolve()
        if not source_path.exists():
            continue
        target_dir.mkdir(parents=True, exist_ok=True)
        target_path = target_dir / source_path.name
        shutil.copy2(source_path, target_path)
        new_rel = f"assets/{slug}/{source_path.name}"
        replacements[raw_path] = new_rel
        if not hero:
            hero = new_rel
    for old, new in replacements.items():
        body = body.replace(f"({old})", f"({new})")
    return body, hero


def build_frontmatter(
    metadata: Dict[str, object],
    slug: str,
    author: str,
    summary: str,
    published_at: str,
    updated_at: str,
    reading_minutes: int,
    tags: List[str],
    hero: str | None,
) -> str:
    data: Dict[str, object] = {
        "title": metadata.get("title", slug.replace("-", " ").title()),
        "slug": slug,
        "author": metadata.get("author", author),
        "summary": summary,
        "publishedAt": published_at,
        "updatedAt": updated_at,
        "readingMinutes": reading_minutes,
    }
    if tags:
        data["tags"] = tags
    if hero:
        data["heroImage"] = hero
    dumped = yaml.safe_dump(data, sort_keys=False).strip()
    return f"---\n{dumped}\n---\n"


def process_file(path: Path, content_root: Path, output_dir: Path, author: str, force: bool) -> None:
    raw = path.read_text(encoding="utf-8")
    metadata, body = split_front_matter(raw)
    slug_source = metadata.get("slug") or path.stem
    slug = slugify(str(slug_source))
    summary = metadata.get("summary") or summarise(body)
    published_at = normalise_date(str(metadata.get("publishedAt") or metadata.get("created") or ""))
    updated_at = normalise_date(str(metadata.get("updatedAt") or metadata.get("updated") or published_at))
    reading_minutes = estimate_reading_minutes(body)
    tags = collect_tags(path, content_root, metadata)

    output_dir.mkdir(parents=True, exist_ok=True)
    output_path = output_dir / f"{slug}.md"
    if output_path.exists() and not force:
        source_mtime = path.stat().st_mtime
        target_mtime = output_path.stat().st_mtime
        if target_mtime >= source_mtime:
            return

    body, hero = rewrite_images(body, path, output_dir / "assets", slug)
    frontmatter = build_frontmatter(metadata, slug, author, str(summary), published_at, updated_at, reading_minutes, tags, hero)
    output_path.write_text(frontmatter + "\n" + body.lstrip("\n"), encoding="utf-8")
    print(f"Wrote {output_path}")


def main() -> None:
    args = parse_args()
    content_dir = args.content_dir.resolve()
    output_dir = args.output_dir.resolve()

    files = sorted(content_dir.rglob("*.md"))
    if not files:
        print(f"No markdown files found under {content_dir}")
        return
    for md_file in files:
        if "/blog/" in md_file.as_posix():
            continue
        process_file(md_file, content_dir, output_dir, args.author, args.force)


if __name__ == "__main__":
    main()
